226 void encodeBlock(const uint8_t* chars, uint8_t* symbols) {
227 int8_t fht[64];
228 for (int t = 0; t < block_len; t++) symbols[t] = 0;
229 for (int b = 0; b < bits; b++) {
230 for (int t = 0; t < block_len; t++) fht[t] = 0;
231 uint8_t ch = encodeChar(chars[b]);
232 if (ch < block_len)
233 fht[ch] = 1;
234 else
235 fht[ch - block_len] = -1;
236 ifht(fht, block_len);
237 int code_bit = (b * shift()) & (block_len - 1);
238 int rotate = 0;
239 for (int t = 0; t < block_len; t++) {
240 if ((scramblingCode() >> code_bit) & 1) fht[t] = -fht[t];
241 code_bit = (code_bit + 1) & (block_len - 1);
242 if (fht[t] < 0) {
243 int bit = b + rotate;
244 if (bit >= bits) bit -= bits;
245 symbols[t] |= 1 << bit;
246 }
247 if (++rotate >= bits) rotate = 0;
248 }
249 }
250 }

255 uint8_t decodeChar(const int8_t* soft, int b, float& peak, float& noise) {
256 int32_t fht[64];
257 int code_bit = (b * shift()) & (block_len - 1);
258 int rotate = b;
259 for (int t = 0; t < block_len; t++) {
260 int32_t bit = soft[t * bits + rotate];
261 if ((scramblingCode() >> code_bit) & 1) bit = -bit;
262 fht[t] = bit;
263 code_bit = (code_bit + 1) & (block_len - 1);
264 if (++rotate >= bits) rotate = 0;
265 }
266 fhtTransform(fht, block_len);
267 int32_t max = 0;
268 int pos = 0;
269 float sum = 0;
270 for (int t = 0; t < block_len; t++) {
271 float v = fht[t];
272 sum += v * v;
273 if (abs(fht[t]) > abs(max)) {
274 max = fht[t];
275 pos = t;
276 }
277 }
278 uint8_t ch = pos;
279 if (max < 0) ch += block_len;
280 peak = abs(max);
281 noise = (sum - peak * peak) / (block_len - 1);
282 return decodeChar(ch);
283 }

328 static void fhtTransform(T* data, int len) {
329 for (int step = 1; step < len; step *= 2)
330 for (int ptr = 0; ptr < len; ptr += 2 * step)
331 for (int j = ptr; j < ptr + step; j++) {
332 T a = data[j], b = data[j + step];
333 data[j] = b + a;
334 data[j + step] = b - a;
335 }

344 T a = data[j], b = data[j + step];
345 data[j] = a - b;
346 data[j + step] = a + b;
