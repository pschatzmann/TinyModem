80 static int encode(const uint8_t* frame, int len, int check_bytes,
81 uint8_t* block, int& block_len) {
82 uint8_t data[BLOCK_SIZE];

88 // the data part is padded with flags, the shortened part is 0
89 memset(data + tg.data, 0, BLOCK_SIZE - tg.data);

92 memcpy(block, codeword, tg.data);

101 static int correct(int t, uint8_t* received) {
102 const Tag& tg = tag(t);
103 uint8_t codeword[BLOCK_SIZE] = {0};
104 memcpy(codeword, received, tg.data);
105 memcpy(codeword + BLOCK_SIZE - tg.check, received + tg.data, tg.check);
106 int result = RSCodec::correct(codeword, BLOCK_SIZE - tg.check, tg.check, 1);

117 static int extract(const uint8_t* data, int len, uint8_t* frame,
118 int frame_size) {
119 uint8_t history = 0;
120 int ones = 0, bits = 0, n = 0;
121 bool in_frame = false;
122 uint8_t current = 0;
123 for (int j = 0; j < len; j++) {
124 for (int b = 0; b < 8; b++) {
125 int bit = (data[j] >> b) & 1;
126 history = (history >> 1) | (bit << 7);
127 if (history == 0x7E) {
128 if (in_frame && n >= AX25Frame::MIN_FRAME + 2) {
129 uint16_t fcs = frame[n - 2] | (frame[n - 1] << 8);
130 return AX25Frame::fcs(frame, n - 2) == fcs ? n - 2 : 0;
131 }
132 in_frame = true;
133 n = bits = ones = 0;
134 current = 0;
135 continue;
136 }
137 if (!in_frame) continue;
138 if (bit) {
139 if (++ones > 6) return 0;
140 } else {
141 if (ones == 5) { // stuffed bit
142 ones = 0;
143 continue;
144 }
145 ones = 0;
146 }
147 current |= bit << bits;
148 if (++bits == 8) {
149 if (n >= frame_size) return 0;
150 frame[n++] = current;
151 current = 0;
152 bits = 0;
153 }
154 }
155 }
156 return 0;
157 }

168 static int stuff(const uint8_t* frame, int len, uint8_t* out, int size) {
169 memset(out, 0, size);
170 int nbits = 0;
171 int max_bits = size * 8;
172 auto put = [&](int bit) {
173 if (nbits >= max_bits) return false;
174 if (bit) out[nbits >> 3] |= 1 << (nbits & 7);
175 nbits++;
176 return true;
177 };
178 for (int b = 0; b < 8; b++) put((0x7E >> b) & 1);
179 uint16_t fcs = AX25Frame::fcs(frame, len);
180 int ones = 0;
181 for (int j = 0; j < len + 2; j++) {
182 uint8_t value = j < len ? frame[j] : (j == len ? fcs & 0xFF : fcs >> 8);
183 for (int b = 0; b < 8; b++) {
184 int bit = (value >> b) & 1;
185 if (!put(bit)) return -1;
186 if (bit) {
187 if (++ones == 5) {
188 if (!put(0)) return -1;
189 ones = 0;
190 }
191 } else {
192 ones = 0;
193 }
194 }
195 }
196 for (int b = 0; b < 8; b++)
197 if (!put((0x7E >> b) & 1)) return -1;
198 int used = (nbits + 7) / 8;
199 // fill with the (rotated) flag pattern
200 for (int b = 0; nbits < max_bits; b++) put((0x7E >> (b & 7)) & 1);
201 return used;
202 }

226 static int encode(const uint8_t* frame, int len, bool max_fec,
227 uint8_t* out) {
228 uint8_t hdr[HEADER_SIZE];

230 const uint8_t* payload = frame + info_pos;
231 int payload_len = len - info_pos;
232 if (info_pos < 0) {
233 // type 0: the complete frame is the payload

235 memset(hdr, 0, sizeof(hdr));

237 setField(hdr, 7, 11, 10, len);

239 payload_len = len;

291 static void scramble(const uint8_t* in, uint8_t* out, int len) {
292 int state = 0x00f;
293 memset(out, 0, len);
294 bool skipping = true;
295 int ob = 0, om = 0x80;
296 auto put = [&](int s) {
297 if (s) out[ob] |= om;
298 om >>= 1;
299 if (om == 0) {
300 om = 0x80;
301 ob++;
302 }
303 };
304 for (int ib = 0; ib < len; ib++) {
305 for (int im = 0x80; im != 0; im >>= 1) {
306 int s = scrambleBit((in[ib] & im) != 0, state);
307 if (ib == 0 && im == 0x04) skipping = false;
308 if (!skipping) put(s);
309 }
310 }
311 for (int n = 0; n < 5; n++) put(scrambleBit(0, state));
312 }

314 static void descramble(const uint8_t* in, uint8_t* out, int len) {
315 int state = 0x1f0;
316 memset(out, 0, len);
317 for (int b = 0; b < len; b++) {
318 for (int m = 0x80; m != 0; m >>= 1) {
319 int bit = (in[b] & m) != 0;
320 int d = (bit ^ state) & 1;
321 state = ((state >> 1) | (bit << 8)) ^ (bit << 3);
322 if (d) out[b] |= m;
323 }
324 }

444 static int type1Header(const uint8_t* frame, int len, bool max_fec,
445 uint8_t* hdr) {
446 memset(hdr, 0, HEADER_SIZE);
447 if (len < 15 || !(frame[13] & 0x01) || (frame[6] & 0x01)) return -1;
448 for (int j = 0; j < 12; j++) {
449 int ch = frame[j < 6 ? j : j + 1] >> 1;
450 if (ch < ' ' || ch > '_') return -1;
451 hdr[j] = ch - ' ';
452 }
453 hdr[12] = (((frame[6] >> 1) & 0x0F) << 4) | ((frame[13] >> 1) & 0x0F);
454 int cbit = (frame[6] & 0x80) ? 1 : 0; // command (or 11)
455 uint8_t c = frame[14];
456 int pf = (c >> 4) & 1;
457 int pos = 15;
458 if ((c & 1) == 0) {
459 // I frame
460 if (len < 16) return -1;
461 int pid = encodePid(frame[15]);
462 if (pid < 0) return -1;
463 setField(hdr, 6, 4, 4, pid);
464 setField(hdr, 6, 11, 7, (pf << 6) | (((c >> 5) & 7) << 3) | ((c >> 1) & 7));
465 pos = 16;
466 } else if ((c & 3) == 1) {
467 // S frame
468 if (len != 15) return -1;
469 setField(hdr, 6, 11, 7,
470 (pf << 6) | (((c >> 5) & 7) << 3) | (cbit << 2) | ((c >> 2) & 3));
471 } else {
472 // U frame
473 int type = -1;
474 for (int j = 0; j < 8; j++)
475 if ((c & ~0x10) == uFrames()[j]) type = j;
476 if (type < 0) return -1;
477 if (type == 5) {
478 // UI
479 if (len < 16) return -1;
480 int pid = encodePid(frame[15]);
481 if (pid < 0) return -1;
482 setField(hdr, 6, 0, 1, 1);
483 setField(hdr, 6, 4, 4, pid);
484 pos = 16;
485 } else {
486 setField(hdr, 6, 4, 4, 1);
487 }
488 setField(hdr, 6, 11, 7, (pf << 6) | (type << 3) | (cbit << 2));
489 }
490 if (len - pos > MAX_PAYLOAD) return -1;
491 setField(hdr, 7, 0, 1, max_fec);
492 setField(hdr, 7, 1, 1, 1);
493 setField(hdr, 7, 11, 10, len - pos);
494 return pos;
495 }

499 static int type1Frame(const uint8_t* hdr, uint8_t* frame) {
500 // callsigns: upper case letters and digits, trailing spaces
501 for (int a = 0; a < 2; a++) {
502 bool end = false;
503 for (int j = 0; j < 6; j++) {
504 char ch = (hdr[a * 6 + j] & 0x3f) + ' ';
505 if (ch == ' ') {
506 end = true;
507 } else if (end || !(isupper((unsigned char)ch) || isdigit((unsigned char)ch))) {
508 return 0;
509 }
510 frame[a * 7 + j] = ch << 1;
511 }
512 }
513 if (frame[0] == (' ' << 1) || frame[7] == (' ' << 1)) return 0;
514 int pid = getField(hdr, 6, 4, 4);
515 int ui = getField(hdr, 6, 0, 1);
516 int control = getField(hdr, 6, 11, 7);
517 int pf = (control >> 6) & 1;
518 bool command = (control & 0x04) != 0;
519 int n = 14;
520 if (pid == 0) {
521 // S frame
522 frame[n++] = (((control >> 3) & 7) << 5) | (pf << 4) | ((control & 3) << 2) | 1;
523 } else if (pid == 1) {
524 // U frame
525 uint8_t c = uFrames()[(control >> 3) & 7] | (pf << 4);
526 frame[n++] = c;
527 if ((c & ~0x10) == 0x03) frame[n++] = 0xf0;
528 } else if (ui) {
529 frame[n++] = 0x03 | (pf << 4);
530 frame[n++] = decodePid(pid);
531 } else {
532 // I frame: always a command
533 command = true;
534 frame[n++] = (((control >> 3) & 7) << 5) | (pf << 4) | ((control & 7) << 1);
535 frame[n++] = decodePid(pid);
536 }
537 frame[6] = 0x60 | (((hdr[12] >> 4) & 0x0F) << 1) | (command ? 0x80 : 0);
538 frame[13] = 0x60 | ((hdr[12] & 0x0F) << 1) | (command ? 0 : 0x80) | 0x01;
539 return n;
540 }
