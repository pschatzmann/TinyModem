146 void process(float* re, float* im) {
147 // bit reversal
148 for (int i = 1, j = 0; i < len; i++) {
149 int bit = len >> 1;
150 for (; j & bit; bit >>= 1) j ^= bit;
151 j ^= bit;
152 if (i < j) {
153 float t = re[i];
154 re[i] = re[j];
155 re[j] = t;
156 t = im[i];
157 im[i] = im[j];
158 im[j] = t;
159 }
160 }
161 for (int size = 2; size <= len; size *= 2) {
162 int half = size / 2;
163 int step = len / size;
164 for (int start = 0; start < len; start += size) {
165 for (int k = 0; k < half; k++) {
166 float c = cos_table[k * step], s = sin_table[k * step];

169 float ti = re[b] * s + im[b] * c;
170 re[b] = re[a] - tr;
171 im[b] = im[a] - ti;
172 re[a] += tr;
173 im[a] += ti;
174 }
175 }
176 }
