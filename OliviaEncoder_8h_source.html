181 void sendSymbol(int tone) {
182 float freq = tone < 0 ? 0 : cfg.toneFreq(tone);
183 uint64_t stop = llround((symbol_index + 1) * symbol_time * cfg.sample_rate);
184 double start_time = symbol_index * symbol_time;
185 for (; sample_index < stop; sample_index++) {
186 float tau = sample_index / (double)cfg.sample_rate - start_time;
187 // raised cosine over two symbol periods: 0 - 1 in the first half
188 float w = 0.5f - 0.5f * cosf((float)M_PI * tau / symbol_time);
189 float value = 0;
190 if (freq > 0) value += w * cosf(phase + 2.0f * (float)M_PI * freq * tau);
191 if (prev_freq > 0)
192 value += (1.0f - w) *
193 cosf(prev_phase +
194 2.0f * (float)M_PI * prev_freq * (tau + symbol_time));
195 samples.write(value);
196 }
197 symbol_index++;
198 // continuous phase in the middle of the symbol and a random +/- 90 degree
199 // step like fldigi
200 prev_freq = freq;
201 prev_phase = phase;
202 random_seed = random_seed * 1103515245u + 12345u;
203 float step = ((random_seed >> 16) & 1) ? (float)M_PI / 2 : -(float)M_PI / 2;
204 phase = fmodf(phase + 2.0f * (float)M_PI * freq * symbol_time + step,
205 2.0f * (float)M_PI);
206 }

210 void sendStartTones() {
211 size_t len = 2048 * cfg.sample_rate / 8000;
212 size_t ramp = len / 8;
213 float freqs[2] = {cfg.center_freq - cfg.bandwidth / 2.0f,
214 cfg.center_freq + cfg.bandwidth / 2.0f};
215 for (int j = 0; j < 4; j++) {
216 float w = 2.0f * (float)M_PI * freqs[j % 2] / cfg.sample_rate;
217 for (size_t i = 0; i < len; i++) {
218 float gain = 1.0f;
219 if (i < ramp) gain = 0.5f - 0.5f * cosf((float)M_PI * i / ramp);
220 if (len - 1 - i < ramp)
221 gain = 0.5f - 0.5f * cosf((float)M_PI * (len - 1 - i) / ramp);
222 samples.write(gain * cosf(w * i));
223 }
224 }
