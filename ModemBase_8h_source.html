20 public:
21 void setFrequency(float frequency) override {
22 this->m_frequency = frequency;

26 T readSample() override {
27 if (this->m_frequency == 0.0f) return 0;
28 float value = this->m_amplitude * sinf(2.0f * (float)M_PI * this->m_cycles);

65 void tone(float freq, size_t nsamples) {
66 if (p_print == nullptr) return;
67 p_generator->setFrequency(freq);
68 int frames_per_buffer = BUFFER_FRAMES * 2 / channels;
69 while (nsamples > 0) {
70 int frames = nsamples < (size_t)frames_per_buffer ? nsamples
71 : frames_per_buffer;
72 int idx = 0;
73 for (int j = 0; j < frames; j++) {
74 int16_t sample = freq == 0.0f ? 0 : scale(p_generator->readSample());
75 for (int ch = 0; ch < channels; ch++) buffer[idx++] = sample;
76 }
77 // blocking write
78 const uint8_t* data = (const uint8_t*)buffer;
79 size_t open = idx * sizeof(int16_t);
80 while (open > 0) {
81 size_t written = p_print->write(data, open);
82 data += written;
83 open -= written;
84 }
85 nsamples -= frames;
86 }
87 }

90 static int32_t amplitude(float volume) {
91 float mag = 32767.0f * volume + 0.5f;
92 if (volume > 1.0f) mag = 32767; // clamp to 1.0 to avoid overflow
93 if (mag < 1) mag = 1; // "short epsilon"
94 return (int)mag;
95 }
96
97 protected:
98 static constexpr int BUFFER_FRAMES = 64;
99 Print* p_print = nullptr;
100 FSKToneGenerator<int16_t> default_generator;
101 audio_tools::SoundGenerator<int16_t>* p_generator = &default_generator;
102 int32_t gain = 32767; // volume in Q15
103 int channels = 1;
104 int16_t buffer[BUFFER_FRAMES * 2];

106 int16_t scale(int16_t sample) {
107 if (gain == 32767) return sample;
108 return (int16_t)((int32_t)sample * gain / 32767);

174 bool begin(int bits_per_sample, int channels, int channel) {
175 if (!(bits_per_sample == 8 || bits_per_sample == 16 ||

186 return false;
187 }
188 bits = bits_per_sample;
189 int bytes = bits / 8;
190 frame_bytes = bytes * channels;
191 sample_offset = bytes * channel;
192 partial_len = 0;
193 return true;
194 }
195
197 void reset() { partial_len = 0; }
198
200 template <class F>
201 void write(const uint8_t* data, size_t len, F on_sample) {
202 size_t j = 0;
203 // complete a frame from the previous call
204 while (partial_len > 0 && j < len) {
205 partial[partial_len++] = data[j++];
206 if (partial_len == frame_bytes) {
207 on_sample(sampleValue(partial + sample_offset));
208 partial_len = 0;
209 }
210 }
211 // complete frames
212 for (; len - j >= (size_t)frame_bytes; j += frame_bytes)
213 on_sample(sampleValue(data + j + sample_offset));
214 // keep the remaining bytes for the next call
215 while (j < len) partial[partial_len++] = data[j++];
216 }
217
218 protected:
219 uint8_t partial[MAX_CHANNELS * 4];
220 int partial_len = 0;
221 int frame_bytes = 2;
222 int sample_offset = 0;
223 int bits = 16;
224
225 float sampleValue(const uint8_t* ptr) {
226 switch (bits) {
227 case 8:
228 return (int8_t)ptr[0] / 128.0f;
229 case 16: {
230 int16_t v;
231 memcpy(&v, ptr, sizeof(v));
232 return v / 32768.0f;
233 }
234 default: {
235 int32_t v;
236 memcpy(&v, ptr, sizeof(v));
237 return v / 2147483648.0f;
238 }
239 }
240 }

253 }
254 int available() { return data.size() - pos; }
255 int peek() { return available() > 0 ? data[pos] : -1; }
256 int read() {
257 int result = peek();
258 if (result < 0) return result;
259 if (++pos == (size_t)data.size()) {
260 clear();
261 } else if (pos >= 64 && pos * 2 >= (size_t)data.size()) {
262 // drop the consumed data
263 size_t n = data.size() - pos;
264 memmove(data.data(), data.data() + pos, n);
265 data.resize(n);
266 pos = 0;
267 }
268 return result;
269 }

272 pos = 0;
