187 void writeKiss(uint8_t ch) {
188 switch (kiss_state) {
189 case KissState::IDLE:
190 if (ch == AX25Frame::KISS_FEND) kiss_state = KissState::COMMAND;
191 break;
192 case KissState::COMMAND:
193 if (ch == AX25Frame::KISS_FEND) break; // repeated FEND
194 // only data frames (command 0) are sent
195 kiss_data_frame = (ch & 0x0F) == 0;
196 input_len = 0;
197 kiss_state = KissState::DATA;
198 break;
199 case KissState::DATA:
200 if (ch == AX25Frame::KISS_FEND) {
201 if (kiss_data_frame && input_len > 0)
202 writeFrame((const uint8_t*)input, input_len);
203 kiss_state = KissState::COMMAND;
204 } else if (ch == AX25Frame::KISS_FESC) {
205 kiss_state = KissState::ESCAPE;
206 } else {
207 addKiss(ch);
208 }
209 break;
210 case KissState::ESCAPE:
211 if (ch == AX25Frame::KISS_TFEND) addKiss(AX25Frame::KISS_FEND);
212 if (ch == AX25Frame::KISS_TFESC) addKiss(AX25Frame::KISS_FESC);
213 kiss_state = KissState::DATA;
214 break;
215 }

243 bool writeFX25(const uint8_t* frame, size_t len) {
244 uint8_t block[FX25::BLOCK_SIZE];
245 int block_len = 0;
246 int tag = FX25::encode(frame, len, cfg.fx25_check_bytes, block, block_len);
247 if (tag == 0) {
248 LOGW("AX25Encoder: frame is too long for FX.25");
249 return false;
250 }
251 startTransmission();
252 uint64_t value = FX25::tag(tag).value;
253 for (int k = 0; k < 8; k++) sendBits((value >> (8 * k)) & 0xFF);
254 for (int j = 0; j < block_len; j++) sendBits(block[j]);
255 sendFlag();
256 frames++;
257 return true;
258 }

261 bool writeIL2P(const uint8_t* frame, size_t len) {
262 // a frame of up to MAX_FRAME bytes needs at most 2 payload blocks
263 uint8_t encoded[IL2P::HEADER_SIZE + IL2P::HEADER_PARITY +
264 AX25Frame::MAX_FRAME + 2 * 16];
265 int n = IL2P::encode(frame, len, cfg.il2p_max_fec, encoded);
266 if (n > 0) {
267 startTransmission();
268 sendRawByte(IL2P::PREAMBLE);
269 for (int k = 2; k >= 0; k--) sendRawByte((IL2P::SYNC_WORD >> (8 * k)) & 0xFF);
270 for (int j = 0; j < n; j++) sendRawByte(encoded[j]);
271 frames++;
272 } else {
273 LOGW("AX25Encoder: frame can't be encoded with IL2P");
274 }
275 return n > 0;
276 }

297 void sendByte(uint8_t value) {
298 for (int b = 0; b < 8; b++) {
299 int bit = (value >> b) & 1;
300 sendBit(bit);
301 if (bit) {

319 void sendTone() {
320 size_t start = llround(bit_index * samples_per_bit);
