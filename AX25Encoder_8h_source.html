179 void writeKiss(uint8_t ch) {
180 switch (kiss_state) {
181 case KissState::IDLE:
182 if (ch == AX25Frame::KISS_FEND) kiss_state = KissState::COMMAND;
183 break;
184 case KissState::COMMAND:
185 if (ch == AX25Frame::KISS_FEND) break; // repeated FEND
186 // only data frames (command 0) are sent
187 kiss_data_frame = (ch & 0x0F) == 0;
188 input_len = 0;
189 kiss_state = KissState::DATA;
190 break;
191 case KissState::DATA:
192 if (ch == AX25Frame::KISS_FEND) {
193 if (kiss_data_frame && input_len > 0)
194 writeFrame((const uint8_t*)input, input_len);
195 kiss_state = KissState::COMMAND;
196 } else if (ch == AX25Frame::KISS_FESC) {
197 kiss_state = KissState::ESCAPE;
198 } else {
199 addKiss(ch);
200 }
201 break;
202 case KissState::ESCAPE:
203 if (ch == AX25Frame::KISS_TFEND) addKiss(AX25Frame::KISS_FEND);
204 if (ch == AX25Frame::KISS_TFESC) addKiss(AX25Frame::KISS_FESC);
205 kiss_state = KissState::DATA;
206 break;
207 }
208 }
