151 static uint16_t fcs(const uint8_t* data, size_t len) {
152 uint16_t crc = 0xFFFF;
153 for (size_t i = 0; i < len; i++) {
154 crc ^= data[i];
155 for (int b = 0; b < 8; b++)
156 crc = (crc & 1) ? (crc >> 1) ^ 0x8408 : (crc >> 1);
157 }
158 return crc ^ 0xFFFF;
159 }

163 static int fromText(const char* text, size_t len, uint8_t* frame) {
164 const char* end = text + len;
165 const char* gt = (const char*)memchr(text, '>', len);
166 const char* colon = (const char*)memchr(text, ':', len);
167 if (gt == nullptr || colon == nullptr || colon < gt) return 0;
168 int n = 0;
169 // destination first, then source, then the digipeaters
170 const char* dest = gt + 1;
171 const char* dest_end = dest;
172 while (dest_end < colon && *dest_end != ',') dest_end++;
173 if (!encodeAddress(dest, dest_end, frame + n, true, false)) return 0;
174 n += 7;
175 if (!encodeAddress(text, gt, frame + n, false, false)) return 0;
176 n += 7;
177 const char* digi = dest_end;
178 while (digi < colon) {
179 digi++; // skip the ','
180 const char* digi_end = digi;
181 while (digi_end < colon && *digi_end != ',') digi_end++;
182 if (n + 7 > 7 * 10) return 0; // max 8 digipeaters
183 if (!encodeAddress(digi, digi_end, frame + n, false, true)) return 0;
184 n += 7;
185 digi = digi_end;
186 }
187 frame[n - 1] |= 0x01; // last address
188 frame[n++] = 0x03; // UI frame
189 frame[n++] = 0xF0; // no layer 3
190 size_t info_len = end - (colon + 1);
191 if (n + info_len > MAX_FRAME) return 0;
192 memcpy(frame + n, colon + 1, info_len);
193 return n + info_len;
194 }

198 static int toText(const uint8_t* frame, size_t len, char* text,
199 size_t text_size) {
200 // find the end of the address field
201 size_t naddr = 0;
202 while (naddr < 10 && (naddr + 1) * 7 <= len) {
203 naddr++;
204 if (frame[naddr * 7 - 1] & 0x01) break;
205 }
206 if (naddr < 2 || !(frame[naddr * 7 - 1] & 0x01)) return 0;
207 size_t pos = naddr * 7;
208 if (pos >= len) return 0;
209 char call[12];
210 size_t n = 0;
211 auto append = [&](const char* str, size_t str_len) {
212 if (n + str_len < text_size) {
213 memcpy(text + n, str, str_len);
214 n += str_len;
215 }
216 };
217 decodeAddress(frame + 7, call); // source
218 append(call, strlen(call));
219 append(">", 1);
220 decodeAddress(frame, call); // destination
221 append(call, strlen(call));
222 // the last repeated digipeater is marked with a '*'
223 int last_repeated = -1;
224 for (size_t j = 2; j < naddr; j++)
225 if (frame[j * 7 + 6] & 0x80) last_repeated = j;
226 for (size_t j = 2; j < naddr; j++) {
227 append(",", 1);
228 decodeAddress(frame + j * 7, call);
229 append(call, strlen(call));
230 if ((int)j == last_repeated) append("*", 1);
231 }
232 append(":", 1);
233 // skip control and PID of UI frames
234 uint8_t control = frame[pos++];
235 if ((control & 0xEF) == 0x03 && pos < len) pos++;
236 append((const char*)frame + pos, len - pos);
237 text[n] = 0;
238 return n;
239 }

242 static size_t writeKiss(Print& out, const uint8_t* frame, size_t len) {
243 size_t n = out.write(KISS_FEND);
244 n += out.write((uint8_t)0x00);
245 for (size_t j = 0; j < len; j++) {
246 if (frame[j] == KISS_FEND) {
247 n += out.write(KISS_FESC);
248 n += out.write(KISS_TFEND);
249 } else if (frame[j] == KISS_FESC) {
250 n += out.write(KISS_FESC);
251 n += out.write(KISS_TFESC);
252 } else {
253 n += out.write(frame[j]);
254 }
255 }
256 n += out.write(KISS_FEND);
257 return n;
258 }

261 static bool encodeAddress(const char* start, const char* end, uint8_t* out,
262 bool command, bool digipeater) {
263 bool repeated = false;
264 if (end > start && end[-1] == '*') {
265 repeated = true;
266 end--;
267 }
268 const char* dash = start;
269 while (dash < end && *dash != '-') dash++;
270 int call_len = dash - start;
271 if (call_len < 1 || call_len > 6) return false;
272 for (int j = 0; j < 6; j++) {
273 char ch = j < call_len ? toupper(start[j]) : ' ';
274 if (!isalnum((unsigned char)ch) && ch != ' ') return false;
275 out[j] = ch << 1;
276 }
277 int ssid = 0;
278 if (dash < end) {
279 ssid = atoi(dash + 1);
280 if (ssid < 0 || ssid > 15) return false;
281 }
282 uint8_t flags = 0x60; // reserved bits
283 if (command || (digipeater && repeated)) flags |= 0x80;
284 out[6] = flags | (ssid << 1);
285 return true;
286 }

288 static void decodeAddress(const uint8_t* in, char* call) {
289 int n = 0;
290 for (int j = 0; j < 6; j++) {
291 char ch = (in[j] >> 1) & 0x7F;
292 if (ch != ' ') call[n++] = ch;
293 }
294 int ssid = (in[6] >> 1) & 0x0F;
295 if (ssid > 0) n += sprintf(call + n, "-%d", ssid);
296 call[n] = 0;
297 }
