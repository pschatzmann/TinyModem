114 static uint16_t fcs(const uint8_t* data, size_t len) {
115 uint16_t crc = 0xFFFF;
116 for (size_t i = 0; i < len; i++) {
117 crc ^= data[i];
118 for (int b = 0; b < 8; b++)
119 crc = (crc & 1) ? (crc >> 1) ^ 0x8408 : (crc >> 1);
120 }
121 return crc ^ 0xFFFF;

126 static int fromText(const char* text, size_t len, uint8_t* frame) {
127 const char* end = text + len;
128 const char* gt = (const char*)memchr(text, '>', len);
129 const char* colon = (const char*)memchr(text, ':', len);
130 if (gt == nullptr || colon == nullptr || colon < gt) return 0;
131 int n = 0;
132 // destination first, then source, then the digipeaters
133 const char* dest = gt + 1;
134 const char* dest_end = dest;
135 while (dest_end < colon && *dest_end != ',') dest_end++;
136 if (!encodeAddress(dest, dest_end, frame + n, true, false)) return 0;
137 n += 7;
138 if (!encodeAddress(text, gt, frame + n, false, false)) return 0;
139 n += 7;
140 const char* digi = dest_end;
141 while (digi < colon) {
142 digi++; // skip the ','
143 const char* digi_end = digi;
144 while (digi_end < colon && *digi_end != ',') digi_end++;
145 if (n + 7 > 7 * 10) return 0; // max 8 digipeaters
146 if (!encodeAddress(digi, digi_end, frame + n, false, true)) return 0;
147 n += 7;
148 digi = digi_end;
149 }
150 frame[n - 1] |= 0x01; // last address
151 frame[n++] = 0x03; // UI frame
152 frame[n++] = 0xF0; // no layer 3
153 size_t info_len = end - (colon + 1);
154 if (n + info_len > MAX_FRAME) return 0;
155 memcpy(frame + n, colon + 1, info_len);
156 return n + info_len;
157 }

161 static int toText(const uint8_t* frame, size_t len, char* text,
162 size_t text_size) {
163 // find the end of the address field
164 size_t naddr = 0;
165 while (naddr < 10 && (naddr + 1) * 7 <= len) {
166 naddr++;
167 if (frame[naddr * 7 - 1] & 0x01) break;
168 }
169 if (naddr < 2 || !(frame[naddr * 7 - 1] & 0x01)) return 0;
170 size_t pos = naddr * 7;
171 if (pos >= len) return 0;
172 char call[12];
173 size_t n = 0;
174 auto append = [&](const char* str, size_t str_len) {
175 if (n + str_len < text_size) {
176 memcpy(text + n, str, str_len);
177 n += str_len;
178 }
179 };
180 decodeAddress(frame + 7, call); // source
181 append(call, strlen(call));
182 append(">", 1);
183 decodeAddress(frame, call); // destination
184 append(call, strlen(call));
185 // the last repeated digipeater is marked with a '*'
186 int last_repeated = -1;
187 for (size_t j = 2; j < naddr; j++)
188 if (frame[j * 7 + 6] & 0x80) last_repeated = j;
189 for (size_t j = 2; j < naddr; j++) {
190 append(",", 1);
191 decodeAddress(frame + j * 7, call);
192 append(call, strlen(call));
193 if ((int)j == last_repeated) append("*", 1);
194 }
195 append(":", 1);
196 // skip control and PID of UI frames
197 uint8_t control = frame[pos++];
198 if ((control & 0xEF) == 0x03 && pos < len) pos++;
199 append((const char*)frame + pos, len - pos);
200 text[n] = 0;
201 return n;
202 }

205 static size_t writeKiss(Print& out, const uint8_t* frame, size_t len) {
206 size_t n = out.write(KISS_FEND);
207 n += out.write((uint8_t)0x00);
208 for (size_t j = 0; j < len; j++) {
209 if (frame[j] == KISS_FEND) {
210 n += out.write(KISS_FESC);
211 n += out.write(KISS_TFEND);
212 } else if (frame[j] == KISS_FESC) {
213 n += out.write(KISS_FESC);
214 n += out.write(KISS_TFESC);
215 } else {
216 n += out.write(frame[j]);
217 }
218 }
219 n += out.write(KISS_FEND);
220 return n;
221 }

224 static bool encodeAddress(const char* start, const char* end, uint8_t* out,
225 bool command, bool digipeater) {
226 bool repeated = false;
227 if (end > start && end[-1] == '*') {
228 repeated = true;
229 end--;
230 }
231 const char* dash = start;
232 while (dash < end && *dash != '-') dash++;
233 int call_len = dash - start;
234 if (call_len < 1 || call_len > 6) return false;
235 for (int j = 0; j < 6; j++) {
236 char ch = j < call_len ? toupper(start[j]) : ' ';
237 if (!isalnum((unsigned char)ch) && ch != ' ') return false;
238 out[j] = ch << 1;
239 }
240 int ssid = 0;
241 if (dash < end) {
242 ssid = atoi(dash + 1);
243 if (ssid < 0 || ssid > 15) return false;
244 }
245 uint8_t flags = 0x60; // reserved bits
246 if (command || (digipeater && repeated)) flags |= 0x80;
247 out[6] = flags | (ssid << 1);
248 return true;
249 }

251 static void decodeAddress(const uint8_t* in, char* call) {
252 int n = 0;
253 for (int j = 0; j < 6; j++) {
254 char ch = (in[j] >> 1) & 0x7F;
255 if (ch != ' ') call[n++] = ch;
256 }
257 int ssid = (in[6] >> 1) & 0x0F;
258 if (ssid > 0) n += sprintf(call + n, "-%d", ssid);
259 call[n] = 0;
260 }
