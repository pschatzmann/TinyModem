231 void addSample(float x) {
232 if (cfg.rx_noise != 0.0f) x += noise() * cfg.rx_noise * 2.0f;
233 // sliding correlation over one bit
234 float values[4] = {x * mark.re, x * mark.im, x * space.re, x * space.im};
235 float* old = products.data() + pos * 4;
236 for (int j = 0; j < 4; j++) {
237 sums[j] += values[j] - old[j];
238 old[j] = values[j];
239 }
240 if (++pos == window) pos = 0;
241 mark.next();
242 space.next();
243 if (++sample_count % 4096 == 0) recalculate();
244
245 float mark_energy = sums[0] * sums[0] + sums[1] * sums[1];
246 float space_energy = sums[2] * sums[2] + sums[3] * sums[3];
247 bool tone = mark_energy > space_energy;
248
249 // digital PLL: a tone change should be in the middle between two bit
250 // samples
251 pll += pll_step;
252 if (tone != last_tone) {
253 pll -= PLL_GAIN * (pll - 0.5f);
254 last_tone = tone;
255 }
256 if (pll >= 1.0f) {
257 pll -= 1.0f;
258 // NRZI: no tone change = 1
259 int bit = tone == bit_tone ? 1 : 0;
260 bit_tone = tone;
261 hdlc(bit);
262 fx25(bit);
263 il2p(tone); // IL2P does not use NRZI
264 }
265 }

268 void recalculate() {
269 for (int j = 0; j < 4; j++) sums[j] = 0;
270 for (int k = 0; k < window; k++)
271 for (int j = 0; j < 4; j++) sums[j] += products[k * 4 + j];
272 mark.normalize();
273 space.normalize();
274 }

276 void hdlc(int bit) {
277 history = (history << 1) | bit;
278 if (history == 0x7E) {
279 // flag: the last 7 bits (0111111) belong to the flag
280 if (in_frame && current_bits == 7) processFrame();
281 in_frame = true;
282 frame_len = 0;
283 current = 0;
284 current_bits = 0;
285 ones = 0;
286 return;
287 }
288 if (!in_frame) return;
289 if (bit) {
290 if (++ones > 6) { // abort
291 in_frame = false;
292 return;
293 }
294 } else {
295 if (ones == 5) { // stuffed bit
296 ones = 0;
297 return;
298 }
299 ones = 0;
300 }
301 // least significant bit first
302 current |= bit << current_bits;
303 if (++current_bits == 8) {
304 if (frame_len < (int)sizeof(frame)) {
305 frame[frame_len++] = current;
306 } else {
307 in_frame = false;
308 }
309 current = 0;
310 current_bits = 0;
311 }
312 }

314 void processFrame() {
315 if (frame_len < AX25Frame::MIN_FRAME + 2) return;
316 int len = frame_len - 2;
317 uint16_t fcs = frame[len] | (frame[len + 1] << 8);
318 if (AX25Frame::fcs(frame, len) != fcs) {
319 errors++;
320 return;
321 }
322 if (fx_tag != 0) {
323 // probably the content of the FX.25 block: we wait for its result
324 deliverPending();
325 memcpy(pending, frame, len);
326 pending_len = len;
327 return;
328 }
329 deliver(frame, len);
330 }

332 void deliverPending() {
333 if (pending_len > 0) deliver(pending, pending_len);
334 pending_len = 0;
335 }

338 void fx25(int bit) {
339 if (fx_tag == 0) {
340 fx_accum = (fx_accum >> 1) | ((uint64_t)bit << 63);
341 int tag = FX25::findTag(fx_accum);
342 if (tag == 0) return;
343 fx_tag = tag;
344 fx_total = FX25::tag(tag).data + FX25::tag(tag).check;
345 fx_len = fx_bits = 0;
346 memset(fx_block, 0, sizeof(fx_block));
347 return;
348 }
349 fx_block[fx_len] |= bit << fx_bits;
350 if (++fx_bits < 8) return;
351 fx_bits = 0;
352 if (++fx_len < fx_total) return;
353 int tag = fx_tag;
354 fx_tag = 0;
355 fx_accum = 0;
356 int corrected = FX25::correct(tag, fx_block);
357 uint8_t result[AX25Frame::MAX_FRAME + 2];
358 int len = corrected < 0 ? 0
359 : FX25::extract(fx_block, FX25::tag(tag).data,
360 result, sizeof(result));
361 if (len <= 0) {
362 deliverPending(); // the standard frame (if any)
363 return;
364 }
365 pending_len = 0; // duplicate
366 fx25_frames++;
367 corrected_bytes += corrected;
368 deliver(result, len);
369 }

372 void il2p(int raw) {
373 il_accum = ((il_accum << 1) | (raw & 1)) & 0xFFFFFF;
374 if (il_state == 0) {
375 if (popcount(il_accum ^ IL2P::SYNC_WORD) <= 1) {
376 il_inverted = false;
377 } else if (popcount((~il_accum & 0xFFFFFF) ^ IL2P::SYNC_WORD) <= 1) {
378 il_inverted = true;
379 } else {
380 return;
381 }
382 il_state = 1;
383 il_len = il_bits = 0;
384 return;
385 }
386 if (++il_bits < 8) return;
387 il_bits = 0;
388 uint8_t value = il_inverted ? ~il_accum & 0xFF : il_accum & 0xFF;
389 if (il_state == 1) {
390 il_raw_header[il_len++] = value;
391 if (il_len < (int)sizeof(il_raw_header)) return;
392 il_corrected = 0;
393 il_need = IL2P::decodeHeader(il_raw_header, il_header, &il_corrected);
394 if (il_need < 0 || il_need > IL2P_MAX_ENCODED) {
395 il_state = 0;
396 return;
397 }
398 il_len = 0;
399 il_state = 2;
400 if (il_need > 0) return;
401 } else {
402 il_payload[il_len++] = value;
403 if (il_len < il_need) return;
404 }
405 il_state = 0;
406 uint8_t result[AX25Frame::MAX_FRAME];
407 int len = IL2P::decodeFrame(il_header, il_payload, result, sizeof(result),
408 &il_corrected);
409 if (len <= 0) return;
410 il2p_frames++;
411 corrected_bytes += il_corrected;
412 deliver(result, len);
413 }

415 static int popcount(uint32_t v) {
416 int n = 0;
417 for (; v; v &= v - 1) n++;
