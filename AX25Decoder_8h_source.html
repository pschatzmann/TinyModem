181 void addSample(float x) {
182 if (cfg.rx_noise != 0.0f) x += noise() * cfg.rx_noise * 2.0f;
183 // sliding correlation over one bit
184 float values[4] = {x * mark.re, x * mark.im, x * space.re, x * space.im};
185 float* old = products.data() + pos * 4;
186 for (int j = 0; j < 4; j++) {
187 sums[j] += values[j] - old[j];
188 old[j] = values[j];
189 }
190 if (++pos == window) pos = 0;
191 mark.next();
192 space.next();
193 if (++sample_count % 4096 == 0) recalculate();
194
195 float mark_energy = sums[0] * sums[0] + sums[1] * sums[1];
196 float space_energy = sums[2] * sums[2] + sums[3] * sums[3];
197 bool tone = mark_energy > space_energy;
198
199 // digital PLL: a tone change should be in the middle between two bit
200 // samples
201 pll += pll_step;
202 if (tone != last_tone) {
203 pll -= PLL_GAIN * (pll - 0.5f);
204 last_tone = tone;
205 }
206 if (pll >= 1.0f) {
207 pll -= 1.0f;
208 // NRZI: no tone change = 1
209 int bit = tone == bit_tone ? 1 : 0;
210 bit_tone = tone;
211 hdlc(bit);
212 }
213 }

216 void recalculate() {
217 for (int j = 0; j < 4; j++) sums[j] = 0;
218 for (int k = 0; k < window; k++)
219 for (int j = 0; j < 4; j++) sums[j] += products[k * 4 + j];
220 mark.normalize();
221 space.normalize();
222 }

224 void hdlc(int bit) {
225 history = (history << 1) | bit;
226 if (history == 0x7E) {
227 // flag: the last 7 bits (0111111) belong to the flag
228 if (in_frame && current_bits == 7) processFrame();
229 in_frame = true;
230 frame_len = 0;
231 current = 0;
232 current_bits = 0;
233 ones = 0;
234 return;
235 }
236 if (!in_frame) return;
237 if (bit) {
238 if (++ones > 6) { // abort
239 in_frame = false;
240 return;
241 }
242 } else {
243 if (ones == 5) { // stuffed bit
244 ones = 0;
245 return;
246 }
247 ones = 0;
248 }
249 // least significant bit first
250 current |= bit << current_bits;
251 if (++current_bits == 8) {
252 if (frame_len < (int)sizeof(frame)) {
253 frame[frame_len++] = current;
254 } else {
255 in_frame = false;
256 }
257 current = 0;
258 current_bits = 0;
259 }
260 }
261
262 void processFrame() {
263 if (frame_len < AX25Frame::MIN_FRAME + 2) return;
264 int len = frame_len - 2;
265 uint16_t fcs = frame[len] | (frame[len + 1] << 8);
266 if (AX25Frame::fcs(frame, len) != fcs) {
267 errors++;
268 return;
269 }
270 frames++;
271 if (p_print == nullptr) return;
272 if (cfg.format == AX25Format::KISS) {
273 AX25Frame::writeKiss(*p_print, frame, len);
274 } else {
275 int n = AX25Frame::toText(frame, len, text, sizeof(text) - 1);
276 if (n > 0) {
277 text[n++] = '\n';
278 p_print->write((const uint8_t*)text, n);
279 }
280 }
