176 static void payload(const uint8_t* data, int len, int ecc, uint8_t* out) {
177 uint8_t msg[CSSConfig::MAX_FRAME + 2];
178 memcpy(msg, data, len);
179 uint16_t crc = crc16(data, len);
180 msg[len] = crc >> 8;
181 msg[len + 1] = crc & 0xFF;
182 RSCodec::encode(msg, len + 2, ecc, out);
183 }

187 static int decodePayload(uint8_t* data, int len, int ecc) {
188 int corrected = RSCodec::correct(data, len + 2, ecc);
189 if (corrected < 0) return -1;
190 uint16_t crc = crc16(data, len);
191 if (data[len] != (crc >> 8) || data[len + 1] != (crc & 0xFF)) return -1;
192 return corrected;
193 }

196 static int toSymbols(const uint8_t* data, int len, int sf, uint16_t* out) {
197 int n = symbols(len, sf);
198 for (int s = 0; s < n; s++) {
199 int value = 0;
200 for (int b = 0; b < sf; b++) {
201 int bit = s * sf + b;
202 int v = bit < len * 8 ? (data[bit / 8] >> (7 - bit % 8)) & 1 : 0;
203 value = (value << 1) | v;
204 }
205 out[s] = inverseGray(value);
206 }
207 return n;
208 }
209
211 static void fromSymbols(const uint16_t* symbols, int n, int sf,
212 uint8_t* data, int len) {
213 memset(data, 0, len);
214 for (int s = 0; s < n; s++) {
215 int value = symbols[s] ^ (symbols[s] >> 1); // Gray code
216 for (int b = 0; b < sf; b++) {
217 int bit = s * sf + b;
218 if (bit >= len * 8) return;
219 if ((value >> (sf - 1 - b)) & 1) data[bit / 8] |= 0x80 >> (bit % 8);
220 }
221 }
222 }
