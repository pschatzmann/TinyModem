58 static int correct(uint8_t* data, int len, int ecc, int fcr = 0) {
59 int n = len + ecc;
60 if (n > MAX_LEN || ecc < 1 || ecc > MAX_ECC) return -1;
61 uint8_t synd[MAX_ECC];
62 if (!syndromes(data, n, ecc, synd, fcr)) return 0; // no errors
63
64 // Berlekamp-Massey: error locator lambda (ascending powers)
65 uint8_t lambda[MAX_ECC + 2] = {1}, prev[MAX_ECC + 2] = {1};
66 uint8_t tmp[MAX_ECC + 2];
67 int nlambda = 1, nprev = 1, L = 0, m = 1;
68 uint8_t b = 1;
69 for (int r = 0; r < ecc; r++) {
70 uint8_t d = synd[r];
71 for (int i = 1; i <= L && i < nlambda; i++)
72 d ^= mul(lambda[i], synd[r - i]);
73 if (d == 0) {
74 m++;
75 continue;
76 }
77 uint8_t coef = div(d, b);
78 bool grow = 2 * L <= r;
79 if (grow) {
80 memcpy(tmp, lambda, nlambda);
81 }
82 int ntmp = nlambda;
83 // lambda -= coef * x^m * prev
84 if (nprev + m > nlambda) {
85 memset(lambda + nlambda, 0, nprev + m - nlambda);
86 nlambda = nprev + m;
87 }
88 for (int i = 0; i < nprev; i++) lambda[i + m] ^= mul(coef, prev[i]);
89 if (grow) {
90 L = r + 1 - L;
91 memcpy(prev, tmp, ntmp);
92 nprev = ntmp;
93 b = d;
94 m = 1;
95 } else {
96 m++;
97 }
98 }
99 if (2 * L > ecc) return -1;
100
101 // Chien search: positions of the errors
102 uint8_t pos[MAX_ECC / 2 + 1];
103 int nerr = 0;
104 for (int i = 0; i < n; i++) {
105 int power = n - 1 - i;
106 uint8_t x_inv = pow2(255 - power); // alpha^-power
107 if (eval(lambda, nlambda, x_inv) == 0) {
108 if (nerr == L) return -1;
109 pos[nerr++] = i;
110 }
111 }
112 if (nerr != L) return -1;
113
114 // Forney: omega = synd * lambda mod x^ecc
115 uint8_t omega[MAX_ECC];
116 for (int i = 0; i < ecc; i++) {
117 uint8_t v = 0;
118 for (int j = 0; j <= i && j < nlambda; j++)
119 v ^= mul(lambda[j], synd[i - j]);
120 omega[i] = v;
121 }
122 for (int k = 0; k < nerr; k++) {
123 int power = n - 1 - pos[k];
124 uint8_t x = pow2(power);
125 uint8_t x_inv = pow2(255 - power);
126 // derivative of lambda: odd coefficients
127 uint8_t deriv = 0;
128 for (int i = 1; i < nlambda; i += 2)
129 deriv ^= mul(lambda[i], powOf(x_inv, i - 1));
130 if (deriv == 0) return -1;
131 // Forney: e = X^(1 - fcr) * omega(X^-1) / lambda'(X^-1)
132 uint8_t factor = pow2(((1 - fcr) * power % 255 + 255) % 255);
133 uint8_t e = mul(factor, div(eval(omega, ecc, x_inv), deriv));
134 data[pos[k]] ^= e;
135 }
136 // the corrected data must be a code word
137 return syndromes(data, n, ecc, synd, fcr) ? -1 : nerr;
138 }

192 static void generator(int ecc, uint8_t* gen, int fcr) {
193 gen[0] = 1;
194 int len = 1;
195 for (int i = 0; i < ecc; i++) {
196 // gen *= (x + alpha^(fcr + i))
197 uint8_t root = pow2(fcr + i);
198 gen[len] = 0;
199 for (int j = len; j > 0; j--) gen[j] ^= mul(gen[j - 1], root);
200 len++;
201 }
202 }
