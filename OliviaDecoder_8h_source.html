64 if (!cfg.isValid()) return false;
65 if (!input.begin(cfg.bits_per_sample, cfg.channels, cfg.channel))
66 return false;
67 converter.begin(cfg.sample_rate, OliviaConfig::BASE_RATE);
68 bits = cfg.bitsPerSymbol();
69 tones = 1 << bits;
70 codec.begin(bits, cfg.contestia);
71 block_len = codec.blockLen();
72 symbol_len = cfg.symbolLen();
73 // the search range is limited by the audio band
74 margin = cfg.sync_margin;
75 int first = cfg.firstCarrier();
76 if (margin > first - 1) margin = first - 1;
77 int upper = symbol_len / 2 - 1 - (first + 2 * tones);
78 if (margin > upper) margin = upper;
79 if (margin < 0) margin = 0;
80 offsets = 2 * margin + 1;
81 phases = SLICES * block_len;
82 first_bin = cfg.firstCarrier() - margin;
83 width = 2 * tones - 1 + 2 * margin;

234 void processSlice() {
235 for (int j = 0; j < symbol_len; j++) {
236 int idx = sample_pos + j;
237 if (idx >= symbol_len) idx -= symbol_len;
238 fft_re[j] = samples[idx] * window[j];
239 fft_im[j] = 0;
240 }
241 fft.process(fft_re.data(), fft_im.data());
242 for (int j = 0; j < width; j++) {
243 int bin = first_bin + j;
244 energy[j] = fft_re[bin] * fft_re[bin] + fft_im[bin] * fft_im[bin];
245 }
246
247 int8_t block_bits[64 * 8];
248 float slice_best = 0;
249 int slice_best_offset = 0;
250 uint64_t slice_block = 0;
251 float slice_noise = 0, slice_snr = 0;
252 for (int o = 0; o < offsets; o++) {
253 int8_t* buffer = soft.data() + (slice * offsets + o) * block_len * bits;
254 softDecode(energy.data() + o, buffer + soft_pos * bits);
255 // decode the block which ends with this symbol
256 linearize(buffer, block_bits);
257 float signal = 0, noise_energy = 0;
258 uint64_t block = 0;
259 for (int b = 0; b < bits; b++) {
260 float peak, noise;
261 uint8_t ch = codec.decodeChar(block_bits, b, peak, noise);
262 block |= (uint64_t)ch << (8 * b);
263 signal += peak;
264 noise_energy += noise;
265 }
266 signal /= bits;
267 noise_energy /= bits;
268 float weight = 1.0f / cfg.sync_integration;
269 float& s = sync_signal[phase * offsets + o];
270 float& n = sync_noise[phase * offsets + o];
271 s += weight * (signal - s);
272 n += weight * (noise_energy - n);
273 if (s > slice_best) {
274 slice_best = s;
275 slice_best_offset = o;
276 slice_block = block;
277 slice_noise = n;
278 slice_snr = signal / sqrtf(noise_energy > 0 ? noise_energy : 1e-9f);
279 }
280 }
281 // steady tones (e.g. the start and stop tones) are not a valid block
282 int8_t* best_buffer =
283 soft.data() + (slice * offsets + slice_best_offset) * block_len * bits;
284 linearize(best_buffer, block_bits);
285 if (repeatedSymbols(block_bits) > block_len / tones + block_len / 4)
286 slice_snr = 0;
287 blocks[phase] = slice_block;
288 block_snr[phase] = slice_snr;
289 if (phase == best_phase || slice_best > best_signal) {
290 best_signal = slice_best;
291 best_phase = phase;
292 best_offset = slice_best_offset;
293 best_noise = slice_noise;
294 }
295 // a block is provided half a block after the best timing
296 int dist = phase - best_phase;
297 if (dist < 0) dist += phases;
298 if (dist == phases / 2) evaluateBlock();
299
300 if (++phase == phases) phase = 0;
301 if (++slice == SLICES) {
302 slice = 0;
303 if (++soft_pos == block_len) soft_pos = 0;
304 }
305 }

308 void softDecode(const float* tone_energy, int8_t* out) {
309 float sum[8] = {0};
310 float total = 0;
311 for (int k = 0; k < tones; k++) {
312 uint8_t symbol = OliviaCodec::binaryCode(k);
313 float e = tone_energy[2 * k];
314 e *= e;
315 total += e;
316 for (int b = 0; b < bits; b++) sum[b] += (symbol >> b) & 1 ? -e : e;
317 }
318 for (int b = 0; b < bits; b++)
319 out[b] = total > 0 ? (int8_t)lroundf(127.0f * sum[b] / total) : 0;
320 }

323 int repeatedSymbols(const int8_t* block) {
324 int result = 0;
325 int prev = -1;
326 for (int t = 0; t < block_len; t++) {
327 int symbol = 0;
328 for (int b = 0; b < bits; b++)
329 if (block[t * bits + b] < 0) symbol |= 1 << b;
330 if (symbol == prev) result++;
331 prev = symbol;
332 }
333 return result;
334 }

337 void linearize(const int8_t* buffer, int8_t* block) {
338 int start = soft_pos + 1; // oldest symbol
339 if (start == block_len) start = 0;
340 int n1 = (block_len - start) * bits;
341 memcpy(block, buffer + start * bits, n1);
342 memcpy(block + n1, buffer, start * bits);
343 }

346 void evaluateBlock() {
347 snr = best_noise > 0 ? best_signal / sqrtf(best_noise)
348 : (best_signal > 0 ? 100.0f : 0.0f);
349 // the integrated timing lags behind a drifting clock: we use the best
350 // block of the neighboring time slices
351 int use = best_phase;
352 for (int d = -1; d <= 1; d += 2) {
353 int p = (best_phase + d + phases) % phases;
354 if (block_snr[p] > block_snr[use]) use = p;
355 }
356 // the block itself must be valid: avoids garbage at the end of a
357 // transmission, while the integrated signal is still strong
358 if (snr < cfg.sync_threshold ||
359 block_snr[use] < BLOCK_THRESHOLD * cfg.sync_threshold) {
360 if (carrier) endOfCarrier();
361 return;
362 }
363 if (!carrier) startOfCarrier();
364 nblocks++;
365 snr_total += snr;
366 uint64_t block = blocks[use];
367 for (int b = 0; b < bits; b++) outputChar((block >> (8 * b)) & 0xFF);
368 }

370 void outputChar(uint8_t ch) {
371 if (!cfg.contestia) {
372 if (escape) {
373 escape = false;
374 ch += 128;

380 if (ch == 0) return; // idle

383 }
