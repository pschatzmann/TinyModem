60 bool begin() override {
61 is_active = false;
62 if (!cfg.isValid()) return false;
63 if (!input.begin(cfg.bits_per_sample, cfg.channels, cfg.channel))
64 return false;
65 n = cfg.chips();
66 m = OVERSAMPLING * n;
67 if (!fft.begin(m)) return false;
68 // baseband with twice the bandwidth as sample rate
69 converter.begin(cfg.sample_rate, OVERSAMPLING * cfg.bandwidth,
70 0.5f / OVERSAMPLING);
71 double w = 2.0 * M_PI * cfg.center_freq / cfg.sample_rate;
72 mix_step_re = cos(w);
73 mix_step_im = -sin(w);
74 ring_re.resize(RING * m);
75 ring_im.resize(RING * m);
76 fft_re.resize(m);
77 fft_im.resize(m);
78 power.resize(m);
79 chirp_re.resize(m);
80 chirp_im.resize(m);
81 for (int k = 0; k < m; k++) {
82 // baseband up chirp from -B/2 to B/2: the phase is the integral of
83 // the frequency
84 double u = (double)k / m; // 0 - 1 within the symbol
85 double ph = 2.0 * M_PI * n * (0.5 * u * u - 0.5 * u);
86 chirp_re[k] = cos(ph);
87 chirp_im[k] = sin(ph);
88 }
89 symbols.resize(CSSFrame::symbols(
90 CSSFrame::payloadBytes(CSSConfig::MAX_FRAME, RSCodec::MAX_ECC), 5));
91 frames = errors = 0;
92 resetState();
93 is_active = true;
94 return true;
95 }

226 void addSample(float x) {
227 if (is_finished) return;
228 float i = x * mix_re, q = x * mix_im;
229 float r = mix_re * mix_step_re - mix_im * mix_step_im;
230 mix_im = mix_re * mix_step_im + mix_im * mix_step_re;
231 mix_re = r;
232 if (++mix_count % 1024 == 0) {
233 float mag = sqrtf(mix_re * mix_re + mix_im * mix_im);
234 mix_re /= mag;
235 mix_im /= mag;
236 }
237 converter.writeComplex(i, q, [this](float re, float im) {
238 addBaseband(re, im);
239 });
240 }

242 void addBaseband(float re, float im) {
243 int idx = count % (RING * m);
244 // factor 2: the mixer provides half of the amplitude
245 ring_re[idx] = 2 * re;
246 ring_im[idx] = 2 * im;

254 Peak spectrum(bool up, float offset) {
255 double w = -2.0 * M_PI * offset / m;
256 float rot_re = cos(w), rot_im = sin(w);
257 float c_re = 1, c_im = 0;
258 for (int k = 0; k < m; k++) {
259 int idx = (next_start + k) % (RING * m);
260 float r_re = ring_re[idx], r_im = ring_im[idx];
261 // up: multiply with conj(chirp), down: with chirp
262 float ch_re = chirp_re[k], ch_im = up ? -chirp_im[k] : chirp_im[k];
263 float y_re = r_re * ch_re - r_im * ch_im;
264 float y_im = r_re * ch_im + r_im * ch_re;
265 fft_re[k] = y_re * c_re - y_im * c_im;
266 fft_im[k] = y_re * c_im + y_im * c_re;
267 float t = c_re * rot_re - c_im * rot_im;
268 c_im = c_re * rot_im + c_im * rot_re;
269 c_re = t;
270 }
271 fft.process(fft_re.data(), fft_im.data());
272 for (int k = 0; k < m; k++)
273 power[k] = fft_re[k] * fft_re[k] + fft_im[k] * fft_im[k];
274 // the part after the wrap around is a tone n bins lower: we add both
275 Peak result;
276 float total = 0, best = 0;
277 for (int k = 0; k < n; k++) {
278 float p = power[k] + power[k + n];
279 total += p;
280 if (p > best) {
281 best = p;
282 result.bin = k;
283 }
284 }
285 result.power = best;
286 float mean = (total - best) / (n - 1);
287 result.ratio = mean > 0 ? best / mean : 0;
288 // fractional position of the stronger tone (Quinn's first estimator,
289 // which is exact for a rectangular window)
290 int k = power[result.bin] >= power[result.bin + n] ? result.bin
291 : result.bin + n;
292 int l = (k + m - 1) % m, r = (k + 1) % m;
293 float pk = power[k];
294 float delta = 0;
295 if (pk > 0) {
296 float a1 = (fft_re[l] * fft_re[k] + fft_im[l] * fft_im[k]) / pk;
297 float a2 = (fft_re[r] * fft_re[k] + fft_im[r] * fft_im[k]) / pk;
298 float d1 = a1 / (1.0f - a1), d2 = -a2 / (1.0f - a2);
299 delta = (d1 > 0 && d2 > 0) ? d2 : d1;
300 if (delta > 0.5f) delta = 0.5f;
301 if (delta < -0.5f) delta = -0.5f;
302 }
303 result.pos = result.bin + delta;
304 return result;
305 }

309 while (pos > n / 2) pos -= n;
310 while (pos <= -n / 2) pos += n;

329 void search() {
330 Peak up = spectrum(true, 0);
331 if (up.ratio >= cfg.detect_threshold) {
332 // the bin may move a little with a different clock of the sender
333 int dist = abs((int)signedBin(up.bin - candidate));
334 hits = hits > 0 && dist <= 2 ? hits + 1 : 1;
335 candidate = up.bin;
336 } else {
337 hits = 0;
338 }
339 if (hits >= PREAMBLE_HITS) {
340 // the window start moves to the symbol start (plus the frequency
341 // offset, which is determined with the down chirps)
342 int shift = -lroundf(OVERSAMPLING * signedBin(up.pos));
343 next_start += m + shift;
344 state = SYNC;
345 nups = 0;
346 sync_windows = 0;
347 last_up = 0;
348 hits = 0;
349 return;
350 }
351 next_start += m;
352 }
353
356 void sync() {
357 Peak up = spectrum(true, 0);
358 Peak down = spectrum(false, 0);
359 if (down.power > up.power && down.ratio >= cfg.detect_threshold) {
360 // a different clock of the sender changes the timing by drift per
361 // symbol: up = tau + offset, down = -(tau + drift) + offset
362 drift = estimateDrift();
363 float p_down = signedBin(down.pos);
364 float tau = (last_up - p_down - drift) / 2;
365 float offset = (last_up + p_down + drift) / 2;
366 // timing of the first down chirp
367 tau += drift;
368 // a clear drift is caused by a different clock: from now on we sample
369 // with the clock of the sender, which also corrects the chirp rate. A
370 // small drift is followed by the tracking.
371 rate_factor = fabsf(drift) > MIN_DRIFT ? 1.0f + drift / n : 1.0f;
372 converter.setRateFactor(rate_factor);
373 drift = 0;
374 // the window moves by whole samples, the rest is part of the frequency
375 // offset (a fractional time offset is harmless with the oversampling)
376 int shift = lroundf(OVERSAMPLING * tau);
377 freq_offset = offset + (OVERSAMPLING * tau - shift) / OVERSAMPLING;
378 sync_offset = freq_offset;
379 // skip the second down chirp
380 next_start += 2 * m - shift;
381 nsymbols = 0;
382 needed = CSSFrame::symbols(CSSFrame::HEADER_BYTES, cfg.spreading_factor);
383 snr_total = 0;
384 state = HEADER;
385 status("### CARRIER css sf%d %.0f Hz @ %.1f Hz ###\n",
386 cfg.spreading_factor, (double)cfg.bandwidth,
387 (double)(cfg.center_freq + frequencyOffset()));
388 return;
389 }
390 last_up = signedBin(up.pos);
391 if (nups < MAX_UPS) ups[nups++] = last_up;
392 next_start += m;
393 if (++sync_windows > cfg.preamble_symbols + 4 ||
394 up.ratio < cfg.detect_threshold) {
395 state = SEARCH;
396 hits = 0;
397 }
398 }

401 float estimateDrift() {
402 if (nups < 2) return 0;
403 float sx = 0, sy = 0, sxx = 0, sxy = 0;
404 for (int j = 0; j < nups; j++) {
405 sx += j;
406 sy += ups[j];
407 sxx += (float)j * j;
408 sxy += j * ups[j];
409 }
410 float d = nups * sxx - sx * sx;
411 return d > 0 ? (nups * sxy - sx * sy) / d : 0;
412 }

415 void data() {
416 Peak peak = spectrum(true, freq_offset);
417 int value = peak.bin;
418 snr_total += peak.ratio;
419 // second order loop: the remaining offset of the peak (frequency error
420 // and the timing drift of a remaining clock difference) is corrected in
421 // the frequency domain. With the oversampling a small time offset of the
422 // window only costs a little energy.
423 float error = signedBin(peak.pos - value);
424 freq_offset += TRACKING_GAIN * error + drift;
425 drift += DRIFT_GAIN * error;
426 // a change of a whole bin since the sync is caused by a timing drift: we
427 // move the window by one chip, which moves the peak by one bin
428 int shift = 0;
429 if (freq_offset - sync_offset > 1.0f) shift = 1;
430 if (freq_offset - sync_offset < -1.0f) shift = -1;
431 freq_offset -= shift;
432 next_start += m - OVERSAMPLING * shift;
433 symbols[nsymbols++] = value;
434 if (nsymbols < needed) return;
435
436 int sf = cfg.spreading_factor;
437 if (state == HEADER) {
438 uint8_t header[CSSFrame::HEADER_BYTES];
439 CSSFrame::fromSymbols(symbols.data(), nsymbols, sf, header,
440 sizeof(header));
441 frame_len = CSSFrame::decodeHeader(header);
442 if (frame_len < 0) {
443 errors++;
444 endOfCarrier(false);
445 return;
446 }
447 state = PAYLOAD;
448 nsymbols = 0;
449 needed = CSSFrame::symbols(
450 CSSFrame::payloadBytes(frame_len, cfg.ecc_bytes), sf);
451 return;
452 }
453 uint8_t payload[CSSConfig::MAX_FRAME + 2 + RSCodec::MAX_ECC];
454 int len = CSSFrame::payloadBytes(frame_len, cfg.ecc_bytes);
455 CSSFrame::fromSymbols(symbols.data(), nsymbols, sf, payload, len);
456 int corrected = CSSFrame::decodePayload(payload, frame_len, cfg.ecc_bytes);
457 if (corrected < 0) {
458 errors++;
459 endOfCarrier(false);
460 return;
461 }
462 frames++;
463 if (p_print != nullptr) p_print->write(payload, frame_len);
464 endOfCarrier(true, corrected);
465 }
466
467 void endOfCarrier(bool ok, int corrected = 0) {
468 rate_factor = 1;
469 converter.setRateFactor(1);
470 int total = nsymbols + (state == PAYLOAD
471 ? CSSFrame::symbols(CSSFrame::HEADER_BYTES,
472 cfg.spreading_factor)
473 : 0);
474 float snr = total > 0 ? 10 * log10f(snr_total / total) : 0;
475 if (ok)
476 status("\n### NOCARRIER ndata=%d corrected=%d snr=%.1f dB ###\n",
477 frame_len, corrected, (double)snr);
478 else
479 status("\n### NOCARRIER frame error snr=%.1f dB ###\n", (double)snr);
480 state = SEARCH;
481 hits = 0;
482 if (cfg.rx_one) is_finished = true;
483 }
