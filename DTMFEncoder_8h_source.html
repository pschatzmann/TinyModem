140 void sendKey(int row, int column) {
141 size_t n = (size_t)cfg.tone_ms * cfg.sample_rate / 1000;
142 size_t ramp = RAMP_MS * cfg.sample_rate / 1000;
143 if (2 * ramp > n) ramp = n / 2;
144 float w1 = 2.0f * (float)M_PI * DTMFKeys::rows()[row] / cfg.sample_rate;
145 float w2 = 2.0f * (float)M_PI * DTMFKeys::columns()[column] / cfg.sample_rate;
146 for (size_t j = 0; j < n; j++) {
147 float gain = 0.5f;
148 if (j < ramp) gain *= 0.5f - 0.5f * cosf((float)M_PI * j / ramp);
149 if (n - j <= ramp)
150 gain *= 0.5f - 0.5f * cosf((float)M_PI * (n - j) / ramp);
151 samples.write(gain * (sinf(w1 * j) + sinf(w2 * j)));
152 }
153 silence(cfg.pause_ms);
154 }
