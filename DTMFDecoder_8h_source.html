189 void addSample(float x, bool add_noise) {
190 if (is_finished) return;
191 if (add_noise && cfg.rx_noise != 0.0f) x += noise() * cfg.rx_noise * 2.0f;
192 for (int k = 0; k < WINDOWS; k++) {
193 Window& w = windows[k];
194 if (w.count++ < 0) continue; // not started yet
195 for (int t = 0; t < TONES; t++) {
196 float s = x + coeff[t] * w.s1[t] - w.s2[t];
197 w.s2[t] = w.s1[t];
198 w.s1[t] = s;
199 }
200 w.power += x * x;
201 if (w.count == window) {
202 decide(detect(w));
203 resetWindow(w);
204 }
205 }
206 if (carrier && held < 0 && ++idle_samples >= timeout) endOfCarrier();
207 }

211 int detect(Window& w) {
212 float amp[TONES];
213 for (int t = 0; t < TONES; t++) {
214 float p = w.s1[t] * w.s1[t] + w.s2[t] * w.s2[t] -
215 coeff[t] * w.s1[t] * w.s2[t];
216 amp[t] = 2.0f * sqrtf(p > 0 ? p : 0) / window;
217 }
218 int row = strongest(amp);
219 int column = strongest(amp + 4);
220 float a_row = amp[row], a_col = amp[4 + column];
221 level = a_row + a_col;
222 if (a_row < cfg.min_amplitude || a_col < cfg.min_amplitude) return -1;
223 if (a_row > twist * a_col || a_col > twist * a_row) return -1;
224 for (int t = 0; t < 4; t++) {
225 if (t != row && amp[t] * amp[t] * group_ratio > a_row * a_row) return -1;
226 if (t != column && amp[4 + t] * amp[4 + t] * group_ratio > a_col * a_col)
227 return -1;
228 }
229 // power of the tones (a^2 / 2) compared to the mean power
230 float tone_power = (a_row * a_row + a_col * a_col) / 2.0f;
231 if (tone_power < ENERGY_RATIO * w.power / window) return -1;
232 return row * 4 + column;
233 }

235 static int strongest(const float* amp) {
236 int result = 0;
237 for (int t = 1; t < 4; t++)
238 if (amp[t] > amp[result]) result = t;

243 void decide(int key) {
244 if (key >= 0 && key == candidate) {
245 hits++;
246 } else {
247 candidate = key;
248 hits = 1;
249 }
250 if (held >= 0 && key != held) {
251 if (++misses >= MISSES) held = -1;
252 } else {
253 misses = 0;
254 }
255 if (candidate >= 0 && hits >= HITS && candidate != held) {
256 held = candidate;
257 misses = 0;
258 bool echo = held == last_key && level < ECHO_LEVEL * last_level;
259 if (!echo) {
260 if (!carrier) startOfCarrier();
261 ndata++;
262 if (p_print != nullptr) p_print->write((uint8_t)currentKey());
263 last_key = held;
264 last_level = 0;
265 }
266 }
267 // the level of the reported key
268 if (held >= 0 && held == last_key && key == held && level > last_level)
269 last_level = level;
270 if (held >= 0) idle_samples = 0;
271 }
