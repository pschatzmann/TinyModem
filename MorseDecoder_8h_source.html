61 bool begin() override {
62 is_active = false;
63 if (!cfg.isValid()) return false;
64 if (!input.begin(cfg.bits_per_sample, cfg.channels, cfg.channel))
65 return false;
66 block = lroundf(cfg.sample_rate * BLOCK_SEC);
67 if (block < 1) block = 1;
68 // window of half a dot (at least one block)
69 window = lroundf(cfg.unitTime() * cfg.sample_rate * WINDOW_UNITS);
70 if (window < block) window = block;
71 history.resize(2 * window);
72 double w = 2.0 * M_PI * cfg.frequency / cfg.sample_rate;
73 step_re = cos(w);
74 step_im = sin(w);

188 void addSample(float x, bool add_noise) {
189 if (is_finished) return;
190 if (add_noise && cfg.rx_noise != 0.0f) x += noise() * cfg.rx_noise * 2.0f;
191 // mix to the baseband and sum over the window
192 float vi = x * re, vq = -x * im;
193 float* old = history.data() + pos * 2;
194 sum_i += vi - old[0];
195 sum_q += vq - old[1];
196 old[0] = vi;
197 old[1] = vq;
198 if (++pos == window) pos = 0;
199 float r = re * step_re - im * step_im;
200 im = re * step_im + im * step_re;
201 re = r;
202 if (++sample_count % 1024 == 0) recalculate();
203 if (++count == block) {
204 count = 0;
205 processBlock(2.0f * sqrtf(sum_i * sum_i + sum_q * sum_q) / window);
206 }
207 }

210 void recalculate() {
211 sum_i = sum_q = 0;
212 for (int k = 0; k < window; k++) {
213 sum_i += history[k * 2];
214 sum_q += history[k * 2 + 1];
215 }
216 float mag = sqrtf(re * re + im * im);
217 re /= mag;
218 im /= mag;
219 }

222 void processBlock(float level) {
223 float block_sec = (float)block / cfg.sample_rate;
224 // signal level: fast attack, slow decay; noise level: while off
225 if (level > signal_level)
226 signal_level += 0.5f * (level - signal_level);
227 else
228 signal_level += 0.002f * (level - signal_level);
229 float threshold = noise_level + 0.5f * (signal_level - noise_level);
230 float hysteresis = 0.1f * (signal_level - noise_level);
231 // a tone must be clearly above the noise (noise alone has peaks)
232 bool raw = tone_on ? level > threshold - hysteresis
233 : level > threshold + hysteresis &&
234 level >= cfg.min_amplitude &&
235 level > MIN_SNR * noise_mean;
236 // the noise level follows the levels without tone (also not while a new
237 // tone is confirmed): it falls fast and rises slowly, so that the end of
238 // a tone hardly raises it
239 if (first_block) {
240 noise_mean = level;
241 first_block = false;
242 }
243 if (!tone_on && !raw) {
244 noise_level += (level < noise_level ? 0.1f : 0.01f) * (level - noise_level);
245 // the average excludes the end and the start of a tone: the window
246 // must have left the last tone and the level must be close to the
247 // noise floor
248 if (duration >= WINDOW_UNITS * unit && level < 2.5f * noise_level)
249 noise_mean += 0.02f * (level - noise_mean);
250 }
251
252 duration += block_sec;
253 if (raw != tone_on) {
254 // the change must last for a while
255 if (++pending * block_sec >= DEBOUNCE_UNITS * unit) {
256 float changed = pending * block_sec;
257 float length = duration - changed;
258 tone_on = raw;
259 pending = 0;
260 duration = changed;
261 if (tone_on)
262 endOfSpace(length);
263 else
264 endOfMark(length);
265 }
266 } else {
267 pending = 0;
268 }
269 // the end of a character, a word and the transmission is detected
270 // during the space
271 if (!tone_on && carrier) checkSpace(duration);
272 }

274 void endOfMark(float raw_length) {
275 const MorseCode::Timing& t = MorseCode::timing(cfg.alphabet);
276 last_mark = raw_length;
277 float length = raw_length + bias;
278 float units = length / unit;
279 char element;
280 float expected;
281 if (units < (t.dot + t.dash) / 2) {
282 element = '.';
283 expected = t.dot;
284 } else if (cfg.alphabet == MorseAlphabet::INTERNATIONAL ||
285 units < (t.dash + t.long_dash) / 2) {
286 element = '-';
287 expected = t.dash;
288 } else if (units < (t.long_dash + t.extra_long_dash) / 2) {
289 element = '_';
290 expected = t.long_dash;
291 } else {
292 element = '=';
293 expected = t.extra_long_dash;
294 }
295 // follow the speed of the sender with the dots and dashes
296 if (cfg.adaptive_speed && (element == '.' || element == '-'))
297 unit += 0.2f * (length / expected - unit);
298 if (code_len < MAX_CODE) code[code_len++] = element;
299 last_element = element;
300 }
301
302 void endOfSpace(float raw_length) {
303 if (!carrier) {
304 startOfCarrier();
305 return;
306 }
307 const MorseCode::Timing& t = MorseCode::timing(cfg.alphabet);
308 float length = raw_length - bias;
309 float units = length / unit;
310 // a dot and an element space have the same length: the difference is
311 // twice the bias
312 if (last_element == '.' && code_len > 0 &&
313 units < (t.element_gap + t.inner_gap + 1) / 2) {
314 bias += 0.2f * ((raw_length - last_mark) / 2 - bias);
315 float max_bias = 0.3f * unit;
316 if (bias > max_bias) bias = max_bias;
317 if (bias < -max_bias) bias = -max_bias;
318 }
319 // a space within an American character
320 if (cfg.alphabet == MorseAlphabet::AMERICAN && code_len > 0 &&
321 units >= (t.element_gap + t.inner_gap) / 2 &&
322 units < (t.inner_gap + t.letter_gap) / 2 && code_len < MAX_CODE)
323 code[code_len++] = ' ';
324 if (word_pending) {
325 output(' ');
326 word_pending = false;
327 }
328 }
329
331 void checkSpace(float raw_length) {
332 const MorseCode::Timing& t = MorseCode::timing(cfg.alphabet);
333 float units = (raw_length - bias) / unit;
334 float letter = cfg.alphabet == MorseAlphabet::AMERICAN
335 ? (t.inner_gap + t.letter_gap) / 2
336 : (t.element_gap + t.letter_gap) / 2;
337 if (code_len > 0 && units >= letter) finishCharacter();
338 if (units >= (t.letter_gap + t.word_gap) / 2 && ndata > 0)
339 word_pending = true;
340 if (units >= cfg.rx_timeout_units) endOfCarrier();
341 }
