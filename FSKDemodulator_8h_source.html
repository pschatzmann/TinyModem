39 float findFrame(const float* samples, unsigned frame_nsamples,
40 unsigned try_first_sample, unsigned try_max_nsamples,
41 unsigned try_step_nsamples, float try_confidence_search_limit,
42 const char* expect_bits_string, uint64_t* bits_outp,
43 float* ampl_outp, unsigned* frame_start_outp) {
44 int expect_n_bits = strlen(expect_bits_string);
45 if (expect_n_bits == 0 || expect_n_bits > MAX_BITS) return 0.0f;
46 float samples_per_bit = (float)frame_nsamples / expect_n_bits;
47
48 unsigned best_t = 0;
49 float best_c = 0.0f, best_a = 0.0f;
50 uint64_t best_bits = 0;
51
52 // Scan the frame positions starting with try_first_sample, alternating
53 // between a step above that, a step below that, above, below, and so on,
54 // until we've scanned the whole try_max_nsamples range.
55 for (int j = 0;; j++) {
56 int up = (j % 2) ? 1 : -1;
57 int t = try_first_sample + up * ((j + 1) / 2) * try_step_nsamples;
58 if (t >= (int)try_max_nsamples) break;
59 if (t < 0) continue;
60
61 float ampl_out = 0.0f;
62 uint64_t bits_out = 0;
63 float c = frameAnalyze(samples + t, samples_per_bit, expect_n_bits,
64 expect_bits_string, &bits_out, &ampl_out);
65 if (best_c < c) {
66 best_t = t;
67 best_c = c;
68 best_a = ampl_out;
69 best_bits = bits_out;
70 // quit searching if confidence > try_confidence_search_limit
71 if (best_c >= try_confidence_search_limit) break;
72 }
73 }
74 *bits_outp = best_bits;
75 *ampl_outp = best_a;
76 *frame_start_outp = best_t;
77 return best_c;
78 }

82 int detectCarrier(const float* samples, unsigned nsamples,

84 float magscalar = 1.0f / ((float)nsamples / 2.0f);
85 float max_mag = 0.0f;

158 void bitAnalyze(const float* samples, unsigned bit_nsamples,
159 unsigned* bit_outp, float* bit_signal_mag_outp,
160 float* bit_noise_mag_outp) {
161 if (detector_nsamples != bit_nsamples) {

163 setupDetector(space_detector, b_space, bit_nsamples);
164 detector_nsamples = bit_nsamples;
165 }
166 float magscalar = 2.0f / (float)bit_nsamples;
167 float mag_mark = magnitude(mark_detector, samples, bit_nsamples) * magscalar;
168 float mag_space =
169 magnitude(space_detector, samples, bit_nsamples) * magscalar;
170 // mark==1, space==0
171 if (mag_mark > mag_space) {
172 *bit_outp = 1;
173 *bit_signal_mag_outp = mag_mark;
174 *bit_noise_mag_outp = mag_space;
175 } else {
176 *bit_outp = 0;
177 *bit_signal_mag_outp = mag_space;
178 *bit_noise_mag_outp = mag_mark;

183 float frameAnalyze(const float* samples, float samples_per_bit, int n_bits,
184 const char* expect_bits, uint64_t* bits_outp,
185 float* ampl_outp) {
186 unsigned bit_nsamples = (float)(samples_per_bit + 0.5f);
187 unsigned bit_values[MAX_BITS];
188 float bit_sig_mags[MAX_BITS];
189 float bit_noise_mags[MAX_BITS];
190
191 // pass #1 - process and check only the "required" (1/0) expect_bits
192 for (int bitnum = 0; bitnum < n_bits; bitnum++) {
193 if (expect_bits[bitnum] == 'd') continue;
194 unsigned bit_begin_sample = (float)(samples_per_bit * bitnum + 0.5f);
195 bitAnalyze(samples + bit_begin_sample, bit_nsamples, &bit_values[bitnum],
196 &bit_sig_mags[bitnum], &bit_noise_mags[bitnum]);
197 if ((unsigned)(expect_bits[bitnum] - '0') != bit_values[bitnum])
198 return 0.0f; // does not match expected; abort frame analysis.
199 }
200
201 // pass #2 - process only the dontcare ('d') expect_bits
202 for (int bitnum = 0; bitnum < n_bits; bitnum++) {
203 if (expect_bits[bitnum] != 'd') continue;
204 unsigned bit_begin_sample = (float)(samples_per_bit * bitnum + 0.5f);
205 bitAnalyze(samples + bit_begin_sample, bit_nsamples, &bit_values[bitnum],
206 &bit_sig_mags[bitnum], &bit_noise_mags[bitnum]);
207 }
208
209 float total_bit_sig = 0.0f, total_bit_noise = 0.0f;
210 float avg_mark_sig = 0.0f, avg_space_sig = 0.0f;
211 unsigned n_mark = 0, n_space = 0;
212 for (int bitnum = 0; bitnum < n_bits; bitnum++) {
213 // Deal with floating point data type quantization noise: if
214 // total_bit_noise <= FLT_EPSILON, assume it to be 0.0 (snr==inf).
215 total_bit_sig += bit_sig_mags[bitnum];
216 if (bit_noise_mags[bitnum] > FLT_EPSILON)
217 total_bit_noise += bit_noise_mags[bitnum];
218 if (bit_values[bitnum] == 1) {
219 avg_mark_sig += bit_sig_mags[bitnum];
220 n_mark++;
221 } else {
222 avg_space_sig += bit_sig_mags[bitnum];
223 n_space++;
224 }
225 }
226
227 // Compute the "frame SNR"
228 float snr = total_bit_sig / total_bit_noise;
229 float avg_bit_sig = total_bit_sig / n_bits;
230 if (n_mark) avg_mark_sig /= n_mark;
231 if (n_space) avg_space_sig /= n_space;
232
233 // Compute average "divergence": bit_mag_divergence / other_bits_mag
234 float divergence = 0.0f;
235 for (int bitnum = 0; bitnum < n_bits; bitnum++) {
236 float avg_bit_sig_other = bit_values[bitnum] ? avg_mark_sig : avg_space_sig;
237 divergence +=
238 fabsf(bit_sig_mags[bitnum] - avg_bit_sig_other) / avg_bit_sig_other;
239 }
240 divergence *= 2;
241 divergence /= n_bits;
242
243 // Frame confidence is the frame ( SNR * consistency )
244 float confidence = snr * (1.0f - divergence);
245 *ampl_outp = avg_bit_sig;
246
247 // least significant bit first
248 *bits_outp = 0;
249 for (int bitnum = 0; bitnum < n_bits && bitnum < 64; bitnum++)
250 *bits_outp |= (uint64_t)bit_values[bitnum] << bitnum;
251
252 return confidence;
253 }
