171 bool resolve() {
172 if (is_resolved) return true;
173 if (baud_rate <= 0.0f) return false;
174 if (data_bits == 0) data_bits = 8;
175
176 if (binary_output || binary_raw_bits) format = DataFormat::BINARY;
177 if (binary_raw_bits) {
178 start_bits = 0;
179 stop_bits = 0;
180 data_bits = binary_raw_bits;
181 }
182
183 if (baud_rate >= 400) {
184 // Bell 202: baud=1200 mark=1200 space=2200
185 autodetect_shift = -(baud_rate * 5 / 6);
186 if (mark_freq == 0) mark_freq = baud_rate / 2 + 600;
187 if (space_freq == 0) space_freq = mark_freq - autodetect_shift;
188 if (bandwidth == 0) bandwidth = 200;
189 } else if (baud_rate >= 100) {
190 // Bell 103: baud=300 mark=1270 space=1070
191 autodetect_shift = 200;
192 if (mark_freq == 0) mark_freq = 1270;
193 if (space_freq == 0) space_freq = mark_freq - autodetect_shift;
194 if (bandwidth == 0) bandwidth = 50;
195 } else {
196 // RTTY: baud=45.45 mark/space=variable shift=-170
197 autodetect_shift = 170;
198 if (mark_freq == 0) mark_freq = 1585;
199 if (space_freq == 0) space_freq = mark_freq - autodetect_shift;
200 if (bandwidth == 0) bandwidth = 10;
201 }
202
203 // defaults: 1 start bit, 1 stop bit
204 if (start_bits < 0) start_bits = 1;
205 if (stop_bits < 0) stop_bits = 1.0f;
206
207 if (frameBits() > 64) return false;
208 if (start_bits > 20) return false;
209
210 // do not transmit any leader tone if no start bits
211 if (start_bits == 0) leader_bits = 0;
212
213 if (inverted) {
214 float t = mark_freq;
215 mark_freq = space_freq;
216 space_freq = t;
217 }
218
219 // restrict bandwidth to <= data rate
220 if (bandwidth > baud_rate) bandwidth = baud_rate;
221
222 // sanitize confidence search limit
223 if (search_limit < confidence) search_limit = confidence;
224
225 is_resolved = true;
226 return true;
227 }
