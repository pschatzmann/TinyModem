101 static bool parseMode(const char* mode, MorseAlphabet& alphabet,
102 float& speed) {
103 if (mode == nullptr) return false;
104 alphabet = MorseAlphabet::INTERNATIONAL;
105 speed = 0;
106 const char* rest;
107 if (startsWith(mode, "morse")) {
108 rest = mode + 5;
109 } else if (startsWith(mode, "cw")) {
110 rest = mode + 2;
111 } else {
112 return false;
113 }
114 if (*rest == '-') rest++;
115 if (startsWith(rest, "american")) {
116 alphabet = MorseAlphabet::AMERICAN;
117 rest += 8;
118 if (*rest == '-') rest++;
119 } else if (startsWith(rest, "international")) {
120 rest += 13;
121 if (*rest == '-') rest++;
122 }
123 if (*rest == 0) return true;
124 char* end;
125 speed = strtol(rest, &end, 10);
126 return end != rest && *end == 0 && speed >= 1 && speed <= 100;
127 }
