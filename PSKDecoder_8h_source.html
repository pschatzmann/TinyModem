213 void addSample(float x, bool add_noise) {
214 if (is_finished) return;
215 if (add_noise && cfg.rx_noise != 0.0f)
216 x += noise() * cfg.rx_noise * 2.0f;
217
218 // mix to the baseband and integrate over one symbol
219 float values[3] = {x * nco_re, -x * nco_im, x * x};
220 float* old = history.data() + pos * 3;
221 sum_i += values[0] - old[0];
222 sum_q += values[1] - old[1];
223 sum_power += values[2] - old[2];
224 old[0] = values[0];
225 old[1] = values[1];
226 old[2] = values[2];
227 if (++pos == window) pos = 0;
228 float r = nco_re * step_re - nco_im * step_im;
229 nco_im = nco_re * step_im + nco_im * step_re;
230 nco_re = r;
231 if (++sample_count % 1024 == 0) recalculate();
232
233 // energy distribution over the symbol period
234 float energy = sum_i * sum_i + sum_q * sum_q;
235 float prev_phase = phase;
236 phase += 1.0f / samples_per_symbol;
237 int bin = (int)(prev_phase * BINS);
238 if (bin >= BINS) bin = BINS - 1;
239 bin_sum[bin] += energy;
240 bin_count[bin]++;
241 if (phase >= 1.0f) {
242 phase -= 1.0f;
243 updateTiming();
244 }
245 // one decision per symbol: the next decision moves towards the time with
246 // the highest energy
247 if (--countdown <= 0) {
248 decide();
249 float error = target - phase; // in symbols
250 if (error > 0.5f) error -= 1.0f;
251 if (error < -0.5f) error += 1.0f;
252 float correction = TIMING_GAIN * error;
253 if (correction > MAX_CORRECTION) correction = MAX_CORRECTION;
254 if (correction < -MAX_CORRECTION) correction = -MAX_CORRECTION;
255 countdown += samples_per_symbol * (1.0f + correction);
256 }
257 }

260 void recalculate() {
261 sum_i = sum_q = sum_power = 0;
262 for (int k = 0; k < window; k++) {
263 sum_i += history[k * 3];
264 sum_q += history[k * 3 + 1];
265 sum_power += history[k * 3 + 2];
266 }
267 float mag = sqrtf(nco_re * nco_re + nco_im * nco_im);
268 nco_re /= mag;
269 nco_im /= mag;
270 }

273 void updateTiming() {
274 float min = 0;
275 for (int b = 0; b < BINS; b++) {
276 if (bin_count[b] > 0) {
277 float value = bin_sum[b] / bin_count[b];
278 hist[b] += TIMING_SMOOTHING * (value - hist[b]);
279 }
280 bin_sum[b] = 0;
281 bin_count[b] = 0;
282 if (b == 0 || hist[b] < min) min = hist[b];
283 }
284 float x = 0, y = 0;
285 for (int b = 0; b < BINS; b++) {
286 float angle = 2.0f * (float)M_PI * (b + 0.5f) / BINS;
287 float weight = hist[b] - min;
288 x += weight * cosf(angle);
289 y += weight * sinf(angle);
290 }
291 if (x == 0 && y == 0) return;
292 float t = atan2f(y, x) / (2.0f * (float)M_PI);
293 target = t < 0 ? t + 1.0f : t;
294 }

300 void decide() {
301 float zi = sum_i, zq = sum_q;
302 float dr = zi * prev_i + zq * prev_q; // z * conj(prev)
303 float di = zq * prev_i - zi * prev_q;
304 prev_i = zi;
305 prev_q = zq;
306 float amplitude = 2.0f * sqrtf(zi * zi + zq * zq) / window;
307 float rms_amplitude = sqrtf(2.0f * sum_power / window);
308 // close to 1 for a carrier, small for noise
309 bool present = amplitude >= cfg.min_amplitude &&
310 amplitude > 0.5f * rms_amplitude;
311 float mag2 = dr * dr + di * di;
312 if (mag2 <= 0) {
313 // digital silence: the signal is lost
314 if (carrier && ++weak >= MAX_WEAK) endOfCarrier();
315 return;
316 }
317 // the mixer provides the conjugate: phase step like fldigi (0 - 2 pi)
318 float raw = atan2f(di, dr);
319 float step = -raw;
320 if (step < 0) step += 2.0f * (float)M_PI;
321 int m = positions();
322 // cos(2 * phase difference): 1 for a clean BPSK signal and the preamble
323 quality += 0.1f * ((dr * dr - di * di) / mag2 - quality);
324 // cos(m * phase difference): QPSK and 8PSK are slowly averaged, because
325 // the FEC of QPSK works with a lot of phase noise
326 quality_m += (m == 2 ? 0.1f : 0.03f) * (cosf(m * raw) - quality_m);
327 int bit = dr > 0 ? 1 : 0;
328
329 if (present && cfg.afc) {
330 // before the carrier is detected we expect the preamble (phase
331 // reversals): this doubles the capture range to half the symbol rate
332 float error;
333 if (!carrier || m == 2) {
334 bool reversal = carrier ? bit == 0 : true;
335 error = reversal ? atan2f(-di, -dr) : atan2f(di, dr);
336 } else {
337 // distance to the nearest phase position
338 float spacing = 2.0f * (float)M_PI / m;
339 error = raw - spacing * lroundf(raw / spacing);
340 }
341 // a steady carrier is not a preamble with a frequency offset
342 if (!carrier && fabsf(error) > 0.75f * (float)M_PI) error = 0;
343 float offset = freq_offset + AFC_GAIN * error * cfg.symbol_rate /
344 (2.0f * (float)M_PI);
345 float max_offset = cfg.symbol_rate / 2;
346 if (offset > max_offset) offset = max_offset;
347 if (offset < -max_offset) offset = -max_offset;
348 setFrequencyOffset(offset);
349 }
350
351 if (!carrier) {
352 // start with a run of phase reversals
353 zeros = (present && bit == 0 && quality > 0.5f) ? zeros + 1 : 0;
354 if (zeros >= MIN_PREAMBLE) startOfCarrier(amplitude);
355 return;
356 }
357
358 // end of the transmission: signal lost or long steady carrier
359 bool weak_symbol = amplitude < cfg.min_amplitude ||
360 amplitude < LOST_LEVEL * track_amplitude;
361 weak = weak_symbol ? weak + 1 : 0;
362 // a steady carrier ends BPSK and 8PSK (QPSK sends data with 0 degrees)
363 bool steady = fabsf(raw) < (float)M_PI / m;
364 ones = steady && cfg.modulation != PSKModulation::QPSK ? ones + 1 : 0;
365 if (ones >= MAX_ONES) nheld = 0; // postamble
366 if (!steady) releaseHeld();
367 float min_quality = cfg.modulation == PSKModulation::BPSK ? 0.2f
368 : cfg.modulation == PSKModulation::QPSK ? 0.03f
369 : 0.1f;
370 if (weak >= MAX_WEAK || ones >= MAX_ONES || quality_m < min_quality) {
371 endOfCarrier();
372 return;
373 }
374 if (!weak_symbol) {
375 track_amplitude += 0.1f * (amplitude - track_amplitude);
376 quality_total += quality_m;
377 amplitude_total += amplitude;
378 nsymbols++;
379 }
380 switch (cfg.modulation) {
381 case PSKModulation::QPSK: {
382 // soft decision: correlation with the 4 possible phase steps
383 float metric[4];
384 for (int sym = 0; sym < 4; sym++) {
385 float expected = (float)M_PI + (float)M_PI / 2 * ((4 - sym) & 3);
386 metric[sym] = cosf(step - expected);
387 }
388 int decoded = qpsk.decode(metric);
389 if (decoded >= 0) varicode(decoded);
390 break;
391 }
392 case PSKModulation::PSK8: {
393 float pos = (step - (float)M_PI) / ((float)M_PI / 4);
394 int sym = ((int)lroundf(pos) % 8 + 8) % 8;
395 for (int b = 0; b < 3; b++) mfskVaricode((sym >> b) & 1);
396 break;
397 }
398 default:
399 varicode(bit);
400 break;
401 }
402 }

405 void mfskVaricode(int bit) {
406 mfsk_code = (mfsk_code << 1) | bit;
407 if ((mfsk_code & 7) != 1) return;
408 uint32_t value = mfsk_code >> 1;
409 int ch = value <= 0xFFFF ? MFSKVaricode::decode(value) : -1;
410 // the character 0 is the idle signal
411 if (ch > 0) {
412 if (ones > 0 && nheld < MAX_ONES) {
413 // a steady carrier sends spaces: the postamble is not provided
414 held[nheld++] = ch;
415 } else {
416 output(ch);
417 }
418 }
419 mfsk_code = 1;
420 }

433 void varicode(int bit) {
434 if (bit == 0 && last_bit == 0) {
435 if ((code >> 1) != 0) {

446 }

450 void startOfCarrier(float amplitude) {
451 carrier = true;

453 code = 0;
454 last_bit = 0;

456 mfsk_code = 0;

461 nsymbols = 0;

463 amplitude_total = 0;
464 const char* name = cfg.modulation == PSKModulation::QPSK ? "qpsk"

466 : "psk";

470
471 void endOfCarrier() {
472 // the bits in the Viterbi decoder
473 if (cfg.modulation == PSKModulation::QPSK)
474 qpsk.flush([this](int bit) { varicode(bit); });
475 releaseHeld();
476 carrier = false;
477 zeros = 0;
478 float n = nsymbols > 0 ? nsymbols : 1;
479 status(
480 "\n### NOCARRIER ndata=%u quality=%.3f ampl=%.3f freq=%.1f Hz ###\n",
481 ndata, (double)(quality_total / n), (double)(amplitude_total / n),
482 (double)(cfg.carrier_freq + freq_offset));
483 setFrequencyOffset(0);
484 if (cfg.rx_one) is_finished = true;
485 }
