201 void addSample(float x, bool add_noise) {
202 if (is_finished) return;
203 if (add_noise && cfg.rx_noise != 0.0f)
204 x += noise() * cfg.rx_noise * 2.0f;
205
206 // mix to the baseband and integrate over one symbol
207 float values[3] = {x * nco_re, -x * nco_im, x * x};
208 float* old = history.data() + pos * 3;
209 sum_i += values[0] - old[0];
210 sum_q += values[1] - old[1];
211 sum_power += values[2] - old[2];
212 old[0] = values[0];
213 old[1] = values[1];
214 old[2] = values[2];
215 if (++pos == window) pos = 0;
216 float r = nco_re * step_re - nco_im * step_im;
217 nco_im = nco_re * step_im + nco_im * step_re;
218 nco_re = r;
219 if (++sample_count % 1024 == 0) recalculate();
220
221 // energy distribution over the symbol period
222 float energy = sum_i * sum_i + sum_q * sum_q;
223 float prev_phase = phase;
224 phase += 1.0f / samples_per_symbol;
225 int bin = (int)(prev_phase * BINS);
226 if (bin >= BINS) bin = BINS - 1;
227 bin_sum[bin] += energy;
228 bin_count[bin]++;
229 if (phase >= 1.0f) {
230 phase -= 1.0f;
231 updateTiming();
232 }
233 // one decision per symbol: the next decision moves towards the time with
234 // the highest energy
235 if (--countdown <= 0) {
236 decide();
237 float error = target - phase; // in symbols
238 if (error > 0.5f) error -= 1.0f;
239 if (error < -0.5f) error += 1.0f;
240 float correction = TIMING_GAIN * error;
241 if (correction > MAX_CORRECTION) correction = MAX_CORRECTION;
242 if (correction < -MAX_CORRECTION) correction = -MAX_CORRECTION;
243 countdown += samples_per_symbol * (1.0f + correction);
244 }
245 }

248 void recalculate() {
249 sum_i = sum_q = sum_power = 0;
250 for (int k = 0; k < window; k++) {
251 sum_i += history[k * 3];
252 sum_q += history[k * 3 + 1];
253 sum_power += history[k * 3 + 2];
254 }
255 float mag = sqrtf(nco_re * nco_re + nco_im * nco_im);
256 nco_re /= mag;
257 nco_im /= mag;
258 }

261 void updateTiming() {
262 float min = 0;
263 for (int b = 0; b < BINS; b++) {
264 if (bin_count[b] > 0) {
265 float value = bin_sum[b] / bin_count[b];
266 hist[b] += TIMING_SMOOTHING * (value - hist[b]);
267 }
268 bin_sum[b] = 0;
269 bin_count[b] = 0;
270 if (b == 0 || hist[b] < min) min = hist[b];
271 }
272 float x = 0, y = 0;
273 for (int b = 0; b < BINS; b++) {
274 float angle = 2.0f * (float)M_PI * (b + 0.5f) / BINS;
275 float weight = hist[b] - min;
276 x += weight * cosf(angle);
277 y += weight * sinf(angle);
278 }
279 if (x == 0 && y == 0) return;
280 float t = atan2f(y, x) / (2.0f * (float)M_PI);
281 target = t < 0 ? t + 1.0f : t;
282 }

285 void decide() {
286 float zi = sum_i, zq = sum_q;
287 float dr = zi * prev_i + zq * prev_q; // z * conj(prev)
288 float di = zq * prev_i - zi * prev_q;
289 prev_i = zi;
290 prev_q = zq;
291 float amplitude = 2.0f * sqrtf(zi * zi + zq * zq) / window;
292 float rms_amplitude = sqrtf(2.0f * sum_power / window);
293 // close to 1 for a carrier, small for noise
294 bool present = amplitude >= cfg.min_amplitude &&
295 amplitude > 0.5f * rms_amplitude;
296 float mag2 = dr * dr + di * di;
297 if (mag2 <= 0) return;
298 // cos(2 * phase difference): 1 for a clean BPSK signal
299 quality += 0.1f * ((dr * dr - di * di) / mag2 - quality);
300 int bit = dr > 0 ? 1 : 0;
301
302 if (present && cfg.afc) {
303 // before the carrier is detected we expect the preamble (phase
304 // reversals): this doubles the capture range to half the symbol rate
305 bool reversal = carrier ? bit == 0 : true;
306 float error = reversal ? atan2f(-di, -dr) : atan2f(di, dr);
307 // a steady carrier is not a preamble with a frequency offset
308 if (!carrier && fabsf(error) > 0.75f * (float)M_PI) error = 0;
309 float offset = freq_offset + AFC_GAIN * error * cfg.symbol_rate /
310 (2.0f * (float)M_PI);
311 float max_offset = cfg.symbol_rate / 2;
312 if (offset > max_offset) offset = max_offset;
313 if (offset < -max_offset) offset = -max_offset;
314 setFrequencyOffset(offset);
315 }
316
317 if (!carrier) {
318 // start with a run of phase reversals
319 zeros = (present && bit == 0 && quality > 0.5f) ? zeros + 1 : 0;
320 if (zeros >= MIN_PREAMBLE) startOfCarrier(amplitude);
321 return;
322 }
323
324 // end of the transmission: signal lost or long steady carrier
325 bool weak_symbol = amplitude < cfg.min_amplitude ||
326 amplitude < LOST_LEVEL * track_amplitude;
327 weak = weak_symbol ? weak + 1 : 0;
328 ones = bit ? ones + 1 : 0;
329 if (weak >= MAX_WEAK || ones >= MAX_ONES || quality < 0.2f) {
330 endOfCarrier();
331 return;
332 }
333 if (!weak_symbol) {
334 track_amplitude += 0.1f * (amplitude - track_amplitude);
335 quality_total += quality;
336 amplitude_total += amplitude;
337 nsymbols++;
338 }
339 varicode(bit);
340 }

343 void varicode(int bit) {
344 if (bit == 0 && last_bit == 0) {
