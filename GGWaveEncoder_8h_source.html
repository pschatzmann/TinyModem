136 void sendMessage() {
137 int len = message_len;
138 int ecc = GGWaveConfig::eccBytes(len);
139 int bpt = protocol.bytes_per_tx;
140 uint8_t plain[1 + GGWaveConfig::MAX_PAYLOAD];
141 plain[0] = len;
142 for (int i = 0; i < len; i++)
143 plain[i + 1] = message[i] ^ (cfg.dss ? GGWaveConfig::dssMagic(i) : 0);
144 memset(encoded, 0, sizeof(encoded));
145 GGWaveRS::encode(plain, 1, GGWaveConfig::LENGTH_BYTES - 1, encoded);
146 GGWaveRS::encode(plain + 1, len, ecc,
147 encoded + GGWaveConfig::LENGTH_BYTES);
148
149 Tone tones[GGWaveConfig::MARKER_TONES];
150 // start marker: the even tones use bin 2i, the odd tones bin 2i + 1
151 for (int i = 0; i < GGWaveConfig::MARKER_TONES; i++)
152 tones[i] = {2 * i + i % 2, i};
153 for (int f = 0; f < GGWaveConfig::MARKER_FRAMES; f++)
154 sendFrame(tones, GGWaveConfig::MARKER_TONES, f,
155 GGWaveConfig::MARKER_FRAMES);
156
157 // data: each nibble selects one of 16 bins
158 int nsymbols = cfg.dataSymbols(len);
159 for (int s = 0; s < nsymbols; s++) {
160 int n = 0;
161 for (int j = 0; j < bpt; j++) {
162 uint8_t value = encoded[s * bpt + j];
163 int k0 = (2 * j) * 16 + (value & 15);
164 int k1 = (2 * j + 1) * 16 + (value >> 4);
165 tones[n++] = {k0, k0 / 2};
166 tones[n++] = {k1, k1 / 2};
167 }
168 for (int f = 0; f < protocol.frames_per_tx; f++)
169 sendFrame(tones, n, f, protocol.frames_per_tx);
170 }
171
172 // end marker: inverse of the start marker
173 for (int i = 0; i < GGWaveConfig::MARKER_TONES; i++)
174 tones[i] = {2 * i + 1 - i % 2, i};
175 for (int f = 0; f < GGWaveConfig::MARKER_FRAMES; f++)
176 sendFrame(tones, GGWaveConfig::MARKER_TONES, f,
177 GGWaveConfig::MARKER_FRAMES);
178 samples.flush();
179 message_len = 0;
180 }

185 void sendFrame(const Tone* tones, int ntones, int cycle, int frames) {
186 const int n = frame_size;
187 const int total = frames * n;
188 const float ds = 0.15f * total;
189 const float ids = 1.0f / ds;
190 const int begin = (int)(0.15f * total);
191 const int end = (int)((1.0f - 0.15f) * total);
192 const float scale = 1.0f / ntones;
193 float sin_phase[GGWaveConfig::MARKER_TONES];
194 float cos_phase[GGWaveConfig::MARKER_TONES];
195 int idx[GGWaveConfig::MARKER_TONES];
196 int step[GGWaveConfig::MARKER_TONES];
197 for (int t = 0; t < ntones; t++) {
198 // ggwave: phase offset = pi * index / data bits per symbol
199 double phase = M_PI * tones[t].phase / (8 * protocol.bytes_per_tx);
200 sin_phase[t] = sin(phase);
201 cos_phase[t] = cos(phase);
202 idx[t] = 0;
203 step[t] = protocol.freq_start + tones[t].bin;
204 }
205 for (int i = 0; i < n; i++) {
206 float value = 0;
207 for (int t = 0; t < ntones; t++) {
208 // sin(angle + phase)
209 value += table.sine(idx[t]) * cos_phase[t] +
210 table.cosine(idx[t]) * sin_phase[t];
211 idx[t] += step[t];
212 if (idx[t] >= n) idx[t] -= n;
213 }
214 int k = cycle * n + i;
215 float envelope = 1.0f;
216 if (k < begin) {
217 envelope = k * ids;
218 } else if (k > end) {
219 envelope = (total - k) * ids;
220 }
221 samples.write(value * envelope * scale);
222 }
223 }
224};
