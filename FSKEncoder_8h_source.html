176 void writeChar(uint8_t ch) {

178 startTransmission();
179 // emit leader tone (mark)

181 tone(idleFreq(), bit_nsamples);
182 }
183 uint32_t words[2];

185
186 if (transmitting < 2) {
187 transmitting = 2;

189 for (int j = 0; j < active_cfg.tx_sync_bytes; j++)
190 transmitFrame(active_cfg.sync_byte, false);
191 }
192 // emit data bits
193 for (int j = 0; j < nwords; j++)
194 transmitFrame(words[j], active_cfg.msb_first);
195 }

197 void transmitFrame(uint64_t bits, bool msb_first) {
198 ModemConfig& c = active_cfg;
199 float mark = c.mark_freq;
200 float space = c.space_freq;
201 if (c.start_bits > 0)
202 tone(c.invert_start_stop ? mark : space, bit_nsamples * c.start_bits);
203 for (int i = 0; i < c.data_bits; i++) {
204 unsigned bit = msb_first ? (bits >> (c.data_bits - i - 1)) & 1
205 : (bits >> i) & 1;
206 tone(bit == 1 ? mark : space, bit_nsamples);
207 }
208 if (c.stop_bits > 0)
209 tone(c.invert_start_stop ? space : mark,
210 (size_t)(bit_nsamples * c.stop_bits));
211 }
