13inline uint64_t bitReverse(uint64_t value, unsigned bits) {
14 uint64_t out = 0;
15 while (bits--) {
16 out = (out << 1) | (value & 1);
17 value >>= 1;
18 }
19 return out;
20}

23inline uint64_t bitWindow(uint64_t value, unsigned offset, unsigned bits) {
24 uint64_t mask = (bits >= 64) ? ~0ULL : ((1ULL << bits) - 1);
25 return (value >> offset) & mask;

76 int encode(uint32_t words[2], uint8_t ch) override {
77 ch = toupper(ch);
78 if (ch >= 0x60) return 0; // non-encodable
79 int n = 0;
80 uint8_t charset_mask = encodeTable(ch)[1];
81 if ((charset & charset_mask) == 0) {
82 if (charset_mask == 0) return 0; // non-encodable
83 if (charset == 0) charset = 1;
84 if (charset_mask != 3) charset = charset_mask;
85 words[n++] = charset == 1 ? LTRS : FIGS;
86 }
87 words[n++] = encodeTable(ch)[0];
88 // TX un-shift on space
89 if (ch == ' ' && usos) charset = 1;
90 return n;
91 }

93 size_t decode(Print& out, uint64_t bits, unsigned /*n_bits*/) override {
94 uint8_t databits = bits & 0x1F;
95 if (databits == FIGS) {
96 charset = 2;
97 return 0;
98 }
99 if (databits == LTRS) {
100 charset = 1;
101 return 0;
102 }
103 // RX un-shift on space
104 if (databits == SPACE && usos) charset = 1;
105 // letters or U.S. figures
106 return out.write(decodeTable(databits)[charset == 1 ? 0 : 1]);
107 }

168class CallerIDCodec : public DataCodec {

175 size_t decode(Print& out, uint64_t bits, unsigned /*n_bits*/) override {
176 if (msg_type == 0) {
177 if (bits == MSG_MDMF || bits == MSG_SDMF)
178 msg_type = bits;
179 else
180 return 0;
181 buf[ndata++] = bits;
182 return 0;
183 }
184 if (ndata >= sizeof(buf)) {
185 reset();
186 return 0;
187 }
188 buf[ndata++] = bits;
189 // collect bytes until we have the message length + type + checksum
190 unsigned msglen = buf[1];
191 if (ndata < msglen + 2) return 0;
192
193 size_t n = out.print("CALLER-ID\n");
194 n += msg_type == MSG_MDMF ? decodeMDMF(out) : decodeSDMF(out);
195 reset();
196 return n;
197 }
198
199 protected:

201 static constexpr uint8_t MSG_SDMF = 0x04;

235 size_t decodeMDMF(Print& out) {
236 size_t n = 0;
237 unsigned i = 0;
238 unsigned msglen = buf[1];
239 const uint8_t* m = buf + 2;
240 while (i < msglen) {
241 unsigned type = *m++;
242 if (type > DATA_NAME_NA) return n; // bad datastream
243 unsigned len = *m++;
244 if (m + 2 + len >= buf + sizeof(buf)) return n; // bad datastream
245 n += printFmt(out, "%-6s ", typeName(type));
246 const char* prdata = nullptr;
247 unsigned prlen = 0;
248 switch (type) {
249 case DATA_DATETIME:
250 if (len >= 8) n += printDateTime(out, m);
251 break;
252 case DATA_PHONE:
253 if (len == 10) {
254 n += printPhone(out, m);
255 break;
256 }
257 // fallthrough
258 case DATA_NAME:
259 prdata = (const char*)m;
260 prlen = len;
261 break;
262 case DATA_PHONE_NA:
263 case DATA_NAME_NA:
264 if (len == 1 && *m == 'O') {
265 prdata = "[N/A]";
266 prlen = 5;
267 } else if (len == 1 && *m == 'P') {
268 prdata = "[blocked]";
269 prlen = 9;
270 }
271 break;
272 default:
273 break;
274 }
275 if (prdata) {
276 n += out.write((const uint8_t*)prdata, prlen);
277 n += out.write('\n');
278 }
279 m += len;
280 i += len + 2;
281 }
282 return n;
283 }

285 size_t decodeSDMF(Print& out) {
286 size_t n = 0;
287 unsigned msglen = buf[1];
288 const uint8_t* m = buf + 2;
289 n += printFmt(out, "%-6s ", typeName(DATA_DATETIME));
290 n += printDateTime(out, m);
291 if (msglen < 8) return n;
292 m += 8;
293 n += printFmt(out, "%-6s ", typeName(DATA_PHONE));
294 unsigned len = msglen - 8;
295 if (len == 10) {
296 n += printPhone(out, m);
297 } else {
298 n += out.write(m, len);
299 n += out.write('\n');
300 }
301 return n;
302 }
