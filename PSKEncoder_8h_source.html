157 void sendSymbol(Shape shape) {
158 size_t start = llround(symbol_index * samples_per_symbol);
159 size_t stop = llround((symbol_index + 1) * samples_per_symbol);
160 symbol_index++;
161 size_t n = stop - start;
162 for (size_t j = 0; j < n; j++) {
163 float t = (float)j / n; // 0 to 1 within the symbol
164 float amplitude;
165 switch (shape) {
166 case Shape::REVERSAL:
167 amplitude = phase_sign * cosf((float)M_PI * t);
168 break;
169 case Shape::FADE_IN:
170 amplitude = phase_sign * sinf((float)M_PI / 2 * t);
171 break;
172 case Shape::FADE_OUT:
173 amplitude = phase_sign * cosf((float)M_PI / 2 * t);
174 break;
175 default:
176 amplitude = phase_sign;
177 break;
178 }
179 samples.write(amplitude * im);
180 nextCarrier();
181 }
182 if (shape == Shape::REVERSAL) phase_sign = -phase_sign;
183 }
184
185 void nextCarrier() {
186 float r = re * step_re - im * step_im;
187 im = re * step_im + im * step_re;
188 re = r;
189 if (++sample_count % 1024 == 0) {
190 float mag = sqrtf(re * re + im * im);
191 re /= mag;
192 im /= mag;
193 }
194 }
