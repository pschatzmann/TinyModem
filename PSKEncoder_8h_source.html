175 void startTransmission() {

177 symbol_index = 0;

187 void sendChar(uint8_t ch) {
188 if (cfg.modulation == PSKModulation::PSK8) {

190 for (int b = c.length - 1; b >= 0; b--) sendBit((c.bits >> b) & 1);

192 }

194 for (int b = c.length - 1; b >= 0; b--) sendBit((c.bits >> b) & 1);
195 sendBit(0);

197 }

199 void sendBit(int bit) {
200 switch (cfg.modulation) {
201 case PSKModulation::QPSK: {
202 int symbol = code.encode(bit);
203 sendStep(((4 - symbol) & 3) * 4);
204 break;
205 }
206 case PSKModulation::PSK8:
207 // the first bit is the least significant bit of the symbol
208 if (bit_count == 0) bits = 0;
209 bits |= bit << bit_count;
210 if (++bit_count == 3) {
211 sendStep(bits * 2);
212 bit_count = 0;
213 }
214 break;
215 default:
216 // 0: phase reversal, 1: same phase
217 sendStep(bit ? STEADY : REVERSAL);
218 break;
219 }
220 }

222 void sendStep(int step) { sendSymbol(step, Shape::TRANSITION); }
223
226 void sendSymbol(int step, Shape shape) {
227 float prev_re = point_re, prev_im = point_im;
228 if (shape == Shape::TRANSITION) {
229 float angle = (float)M_PI + step * (float)M_PI / 8;
230 float c = cosf(angle), s = sinf(angle);
231 // round to the exact values of the 8 main positions
232 if (fabsf(c) < 1e-6f) c = 0;
233 if (fabsf(s) < 1e-6f) s = 0;
234 point_re = prev_re * c - prev_im * s;
235 point_im = prev_re * s + prev_im * c;
236 }
237 size_t start = llround(symbol_index * samples_per_symbol);
238 size_t stop = llround((symbol_index + 1) * samples_per_symbol);
239 symbol_index++;
240 size_t n = stop - start;
241 for (size_t j = 0; j < n; j++) {
242 float t = (float)j / n; // 0 to 1 within the symbol
243 float i_value, q_value;
244 if (shape == Shape::TRANSITION) {
245 // cosine shaped transition from the previous point
246 float a = 0.5f + 0.5f * cosf((float)M_PI * t);
247 i_value = a * prev_re + (1.0f - a) * point_re;
248 q_value = a * prev_im + (1.0f - a) * point_im;
249 } else {
250 float a = shape == Shape::FADE_IN ? sinf((float)M_PI / 2 * t)
251 : cosf((float)M_PI / 2 * t);
252 i_value = a * point_re;
253 q_value = a * point_im;
254 }
255 // like fldigi: I * cos + Q * sin
256 samples.write(i_value * re + q_value * im);
257 nextCarrier();
258 }
259 }
260
261 void nextCarrier() {
262 float r = re * step_re - im * step_im;
263 im = re * step_im + im * step_re;
264 re = r;
265 if (++sample_count % 1024 == 0) {
266 float mag = sqrtf(re * re + im * im);
267 re /= mag;
268 im /= mag;
269 }
270 }
