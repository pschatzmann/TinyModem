135 void sendFrame() {
136 int n = cfg.carriers;
137 // sync symbol: even carriers with known phases
138 for (int i = 0; i < n; i++) phases[i] = OFDMConfig::knownPhase(0, i);
139 sendSymbol(true);
140 // reference symbol: all carriers with known phases
141 for (int i = 0; i < n; i++) phases[i] = OFDMConfig::knownPhase(1, i);
142 sendSymbol(false);
143 // header: length and inverted length (16 bit each)
144 sendHeader((uint32_t)frame_len << 16 | (uint16_t)~frame_len);
145 // data
146 int bit_count = 0;
147 sendBits(frame.data(), frame_len, bit_count);
148 if (bit_count > 0) sendSymbol(false);
149 samples.flush();
150 frame_len = 0;
151 }

157 void sendBits(const uint8_t* data, int len, int& bit_count) {
158 int n = cfg.carriers;
159 for (int j = 0; j < len; j++) {
160 for (int b = 6; b >= 0; b -= 2) {
161 int dibit = (data[j] >> b) & 3;
162 int carrier = bit_count / 2;
163 phases[carrier] = (phases[carrier] + grayToPhase(dibit)) & 3;
164 bit_count += 2;
165 if (bit_count == 2 * n) {
166 sendSymbol(false);
167 bit_count = 0;
168 }
169 }
170 }
171 }

175 void sendHeader(uint32_t header) {
176 int symbols = OFDMConfig::headerSymbols(cfg.carriers);
177 for (int s = 0; s < symbols; s++) {
178 for (int i = 0; i < cfg.carriers; i++) {
179 int bit = (header >> (31 - (s * cfg.carriers + i) % 32)) & 1;
180 if (bit) phases[i] = (phases[i] + 2) & 3;
181 }
182 sendSymbol(false);
183 }

193 void sendSymbol(bool sync) {
194 int n = cfg.carriers;
195 float a = sync ? amplitude * 1.41421356f : amplitude;
196 for (int t = -prefix_size; t < fft_size; t++) {
197 int time = t < 0 ? t + fft_size : t;
198 float value = 0;
199 for (int i = 0; i < n; i++) {
200 int bin = first_bin + i;
201 if (sync && (bin & 1)) continue; // only even carriers
202 int idx = (int)(((int64_t)bin * time) % fft_size);
203 // cos(angle + phase * 90 degrees)
204 switch (phases[i]) {
205 case 0:
206 value += table.cosine(idx);
207 break;
208 case 1:
209 value -= table.sine(idx);
210 break;
211 case 2:
212 value -= table.cosine(idx);
213 break;
214 default:
215 value += table.sine(idx);
216 break;
217 }
218 }
219 samples.write(a * value);
220 }
221 }
