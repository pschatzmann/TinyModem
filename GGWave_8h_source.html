64 static void encode(const uint8_t* msg, int len, int ecc, uint8_t* out) {
65 uint8_t gen[MAX_LEN + 1];
66 generator(ecc, gen);
67 uint8_t work[MAX_LEN];
68 memcpy(work, msg, len);
69 memset(work + len, 0, ecc);
70 // polynomial division: the remainder are the parity bytes
71 for (int i = 0; i < len; i++) {
72 uint8_t coef = work[i];
73 if (coef == 0) continue;
74 for (int j = 1; j <= ecc; j++) work[i + j] ^= mul(gen[j], coef);
75 }
76 memcpy(out, msg, len);
77 memcpy(out + len, work + len, ecc);
78 }

82 static bool decode(uint8_t* data, int len, int ecc) {
83 int n = len + ecc;
84 if (n > MAX_LEN || ecc < 1) return false;
85 uint8_t synd[MAX_LEN];
86 if (!syndromes(data, n, ecc, synd)) return true; // no errors
87
88 // Berlekamp-Massey: error locator lambda (ascending powers)
89 uint8_t lambda[MAX_LEN + 1] = {1}, prev[MAX_LEN + 1] = {1};
90 uint8_t tmp[MAX_LEN + 1];
91 int nlambda = 1, nprev = 1, L = 0, m = 1;
92 uint8_t b = 1;
93 for (int r = 0; r < ecc; r++) {
94 uint8_t d = synd[r];
95 for (int i = 1; i <= L && i < nlambda; i++)
96 d ^= mul(lambda[i], synd[r - i]);
97 if (d == 0) {
98 m++;
99 continue;
100 }
101 uint8_t coef = div(d, b);
102 bool grow = 2 * L <= r;
103 if (grow) {
104 memcpy(tmp, lambda, nlambda);
105 }
106 int ntmp = nlambda;
107 // lambda -= coef * x^m * prev
108 if (nprev + m > nlambda) {
109 memset(lambda + nlambda, 0, nprev + m - nlambda);
110 nlambda = nprev + m;
111 }
112 for (int i = 0; i < nprev; i++) lambda[i + m] ^= mul(coef, prev[i]);
113 if (grow) {
114 L = r + 1 - L;
115 memcpy(prev, tmp, ntmp);
116 nprev = ntmp;
117 b = d;
118 m = 1;
119 } else {
120 m++;
121 }
122 }
123 if (2 * L > ecc) return false;
124
125 // Chien search: positions of the errors
126 int pos[MAX_LEN];
127 int nerr = 0;
128 for (int i = 0; i < n; i++) {
129 int power = n - 1 - i;
130 uint8_t x_inv = pow2(255 - power); // alpha^-power
131 if (eval(lambda, nlambda, x_inv) == 0) {
132 if (nerr == L) return false;
133 pos[nerr++] = i;
134 }
135 }
136 if (nerr != L) return false;
137
138 // Forney: omega = synd * lambda mod x^ecc
139 uint8_t omega[MAX_LEN];
140 for (int i = 0; i < ecc; i++) {
141 uint8_t v = 0;
142 for (int j = 0; j <= i && j < nlambda; j++)
143 v ^= mul(lambda[j], synd[i - j]);
144 omega[i] = v;
145 }
146 for (int k = 0; k < nerr; k++) {
147 int power = n - 1 - pos[k];
148 uint8_t x = pow2(power);
149 uint8_t x_inv = pow2(255 - power);
150 // derivative of lambda: odd coefficients
151 uint8_t deriv = 0;
152 for (int i = 1; i < nlambda; i += 2)
153 deriv ^= mul(lambda[i], powOf(x_inv, i - 1));
154 if (deriv == 0) return false;
155 uint8_t e = mul(x, div(eval(omega, ecc, x_inv), deriv));
156 data[pos[k]] ^= e;
157 }
158 // the corrected data must be a code word
159 return !syndromes(data, n, ecc, synd);
160 }

163 struct Tables {
164 uint8_t exp[512];
165 uint8_t log[256];
166 Tables() {
167 int x = 1;
168 for (int i = 0; i < 255; i++) {
169 exp[i] = x;
170 log[x] = i;
171 x <<= 1;
172 if (x & 0x100) x ^= 0x11d;
173 }
174 for (int i = 255; i < 512; i++) exp[i] = exp[i - 255];
175 log[0] = 0;
176 }
177 };

192 const Tables& t = tables();
193 return t.exp[(t.log[a] + 255 - t.log[b]) % 255];

197 static uint8_t pow2(int power) { return tables().exp[power % 255]; }

199 static uint8_t powOf(uint8_t x, int power) {
200 if (power == 0) return 1;
201 if (x == 0) return 0;
202 const Tables& t = tables();
203 return t.exp[(t.log[x] * power) % 255];
204 }

207 static uint8_t eval(const uint8_t* poly, int n, uint8_t x) {
208 uint8_t y = 0;
209 for (int i = n - 1; i >= 0; i--) y = mul(y, x) ^ poly[i];
210 return y;
211 }

214 static void generator(int ecc, uint8_t* gen) {
215 gen[0] = 1;
216 int len = 1;
217 for (int i = 0; i < ecc; i++) {
218 // gen *= (x + alpha^i)
219 uint8_t root = pow2(i);
220 gen[len] = 0;
221 for (int j = len; j > 0; j--) gen[j] ^= mul(gen[j - 1], root);
222 len++;
223 }
224 }
225
228 static bool syndromes(const uint8_t* data, int n, int ecc, uint8_t* synd) {
229 bool errors = false;
230 for (int j = 0; j < ecc; j++) {
231 uint8_t x = pow2(j);
232 uint8_t y = 0;
233 for (int i = 0; i < n; i++) y = mul(y, x) ^ data[i];
234 synd[j] = y;
235 if (y != 0) errors = true;
236 }
237 return errors;
238 }
