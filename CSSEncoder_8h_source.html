135
136 symbol_index = 0;
137 phase = 0;
138 written = 0;
139 int nsymbols = cfg.preamble_symbols + 2 + nh + np;
140 total = llround(nsymbols * (double)cfg.symbolTime() * cfg.sample_rate);
141 ramp = RAMP_SEC * cfg.sample_rate;
142 for (int j = 0; j < cfg.preamble_symbols; j++) sendChirp(0, true);
143 for (int j = 0; j < 2; j++) sendChirp(0, false);
144 for (int j = 0; j < nh; j++) sendChirp(header_symbols[j], true);
145 for (int j = 0; j < np; j++) sendChirp(payload_symbols[j], true);

152 static double sawIntegral(double x) {
153 double f = floor(x);
154 return f / 2 + (x - f) * (x - f) / 2;
155 }
156
160 void sendChirp(int value, bool up) {
161 double t_sym = cfg.symbolTime();
162 double t_start = symbol_index * t_sym;
163 uint64_t start = llround(t_start * cfg.sample_rate);
164 uint64_t stop = llround((symbol_index + 1) * t_sym * cfg.sample_rate);
165 symbol_index++;
166 double offset = (double)value / cfg.chips();
167 double cycles_per_unit = cfg.bandwidth * t_sym; // B * T
168 for (uint64_t j = start; j < stop; j++) {
169 double t = j / (double)cfg.sample_rate - t_start;
170 double u = t / t_sym; // 0 - 1 within the symbol
171 // phase = 2 pi * integral of (f0 + B * frac(...)) dt
172 double sweep = up ? sawIntegral(offset + u) - sawIntegral(offset)
173 : -sawIntegral(-u);
174 double ph = phase + 2.0 * M_PI * (cfg.lowFreq() * t + cycles_per_unit * sweep);
175 float gain = 1.0f;
176 if (written < ramp) gain = 0.5f - 0.5f * cosf((float)M_PI * written / ramp);
177 if (total - written <= ramp)
178 gain = 0.5f - 0.5f * cosf((float)M_PI * (total - written) / ramp);
179 samples.write(gain * (float)sin(ph));
180 written++;
181 }
182 // phase at the end of the symbol: the mean frequency is the center
183 phase = fmod(phase + 2.0 * M_PI * cfg.center_freq * t_sym, 2.0 * M_PI);
184 }
