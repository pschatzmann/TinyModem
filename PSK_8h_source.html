302class QPSKCode {
303 public:
304 static constexpr int K = 5;
305 static constexpr int STATES = 1 << (K - 1);
307 static constexpr int DEPTH = 32;
308
310 static int output(int reg) {
311 return parity(reg & 0x17) | (parity(reg & 0x19) << 1);
312 }
313
315 int encode(int bit) {
316 shreg = ((shreg << 1) | (bit & 1)) & ((1 << K) - 1);
317 return output(shreg);
318 }
319
321 void reset() {
322 shreg = 0;
323 for (int s = 0; s < STATES; s++) metrics[s] = 0;
324 count = 0;
325 pos = 0;
326 }
327
330 int decode(const float metric[4]) {
331 float next[STATES];
332 uint16_t decisions = 0;
333 for (int s = 0; s < STATES; s++) {
334 // predecessors: the oldest bit (msb of the predecessor) was 0 or 1
335 int bit = s & 1;
336 int p0 = s >> 1, p1 = (s >> 1) | (STATES >> 1);
337 float m0 = metrics[p0] + metric[output((p0 << 1) | bit)];
338 float m1 = metrics[p1] + metric[output((p1 << 1) | bit)];
339 if (m1 > m0) {
340 next[s] = m1;
341 decisions |= 1 << s;
342 } else {
343 next[s] = m0;
344 }
345 }
346 float best = next[0];
347 for (int s = 1; s < STATES; s++)
348 if (next[s] > best) best = next[s];
349 for (int s = 0; s < STATES; s++) metrics[s] = next[s] - best;
350 history[pos] = decisions;
351 pos = (pos + 1) % DEPTH;
352 if (count < DEPTH) count++;
353 return count < DEPTH ? -1 : traceBack(DEPTH);
354 }
355
357 template <class F>
358 void flush(F on_bit) {
359 // the bit of DEPTH steps ago was already provided
360 int n = count == DEPTH ? DEPTH - 1 : count;
361 for (int j = n; j > 0; j--) on_bit(traceBack(j));
362 count = 0;
363 }
364
365 protected:
366 int shreg = 0;
367 float metrics[STATES] = {0};
368 uint16_t history[DEPTH];
369 int pos = 0;
370 int count = 0;
371
372 static int parity(int v) {
373 v ^= v >> 4;
374 v ^= v >> 2;
375 v ^= v >> 1;
376 return v & 1;
377 }
378
380 int traceBack(int depth) {
381 int state = 0;
382 for (int s = 1; s < STATES; s++)
383 if (metrics[s] > metrics[state]) state = s;
384 int idx = pos;
385 for (int j = 0; j < depth - 1; j++) {
386 idx = (idx + DEPTH - 1) % DEPTH;
387 int msb = (history[idx] >> state) & 1;
388 state = (state >> 1) | (msb << (K - 2));
389 }
390 return state & 1;
391 }
392};
