253 void addSample(float x, bool add_noise) {
254 if (is_finished) return;
255 if (add_noise && cfg.rx_noise != 0.0f)
256 x += noise() * cfg.rx_noise * 2.0f;
257 // DFT of the block: the index of the sample is counted modulo the frame
258 // size, so the blocks of a frame can be added
259 const int n = frame_size;
260 for (int b = 0; b < nbins; b++) {
261 int idx = table_idx[b];
262 block_re[b] += x * table.cosine(idx);
263 block_im[b] -= x * table.sine(idx);
264 idx += freq_start + b;
265 if (idx >= n) idx -= n;
266 table_idx[b] = idx;
267 }
268 if (++sample_pos == block_size) {
269 sample_pos = 0;
270 endBlock();
271 }
272 }

280 void endBlock() {
281 uint32_t t = block_no;
282 float* re = ringRe(t);
283 float* im = ringIm(t);
284 for (int b = 0; b < nbins; b++) {
285 re[b] = block_re[b];
286 im[b] = block_im[b];
287 block_re[b] = block_im[b] = 0;
288 }
289 // power in the window of the last frame
290 float* power = ringPower(t);
291 float score = 0;
292 if (t + 1 >= (uint32_t)WINDOW) {
293 for (int b = 0; b < nbins; b++) {
294 float sum_re = 0, sum_im = 0;
295 for (int k = 0; k < WINDOW; k++) {
296 sum_re += ringRe(t - k)[b];
297 sum_im += ringIm(t - k)[b];
298 }
299 power[b] = sum_re * sum_re + sum_im * sum_im;
300 }
301 score = markerScore(power, nbins);
302 } else {
303 for (int b = 0; b < nbins; b++) power[b] = 0;
304 }
305 scores[t % SCORE_RING] = score;
306 trackStartMarker(score >= detect_score);
307 block_no++;
308 processCandidates(t);
309 }

315 float markerScore(const float* power, int n) {
316 float max_power = 0;
317 for (int b = 0; b < n; b++)
318 if (power[b] > max_power) max_power = power[b];
319 float floor = max_power * 1e-4f;
320 if (floor <= 0) return 0;
321 float score = 0;
322 for (int i = 0; i < GGWaveConfig::MARKER_TONES; i++) {
323 float on = power[2 * i + i % 2];
324 float off = power[2 * i + 1 - i % 2];
325 float sum = on + off;
326 if (sum > floor) score += (on - off) / sum;
327 }
328 return score;
329 }

331 void trackStartMarker(bool detected) {
332 uint32_t t = block_no;
333 if (detected) {
334 if (!run_active) {
335 run_active = true;
336 run_first = t;
337 run_detections = 0;
338 }
339 run_detections++;
340 run_misses = 0;
341 return;
342 }
343 if (!run_active || ++run_misses <= MAX_MISSES) return;
344 run_active = false;
345 if (run_detections < MIN_DETECTIONS) return;
346 // find the windows which are completely inside the marker: they have
347 // the highest sum of the scores
348 uint32_t oldest = t + 1 > (uint32_t)SCORE_RING ? t + 1 - SCORE_RING : 0;
349 uint32_t from = run_first > oldest + 2 * BLOCKS ? run_first - 2 * BLOCKS
350 : oldest;
351 if (t + 1 < from + MARKER_WINDOWS) return;
352 uint32_t to = t + 1 - MARKER_WINDOWS;
353 float sum = 0;
354 for (int k = 0; k < MARKER_WINDOWS; k++) sum += scores[(from + k) % SCORE_RING];
355 float best = sum;
356 uint32_t best_pos = from;
357 for (uint32_t pos = from + 1; pos <= to; pos++) {
358 sum += scores[(pos + MARKER_WINDOWS - 1) % SCORE_RING] -
359 scores[(pos - 1) % SCORE_RING];
360 if (sum > best) {
361 best = sum;
362 best_pos = pos;
363 }
364 }
365 // the first window inside the marker ends in its block WINDOW - 1
366 startCandidates(best_pos + MARKER_WINDOWS);
367 }

369 void startCandidates(uint32_t data_start) {
370 if (nalive > 0) failed();
371 // the oldest block which is still in the history
372 uint32_t oldest = block_no + 1 > (uint32_t)RING ? block_no + 1 - RING : 0;
373 int idx = 0;

375 for (int k = 0; k < NOFFSETS; k++) {

406 void processCandidates(uint32_t t) {
407 for (int j = 0; j < (int)candidates.size() && nalive > 0; j++) {

417 void processBlock(Candidate& c, uint32_t t) {
418 int symbol_blocks = c.protocol.frames_per_tx * BLOCKS;
419 uint32_t rel = t - c.start;
420 if (!c.data_done) {
421 int within = rel % symbol_blocks;
422 // windows inside the symbol without its first and last block
423 // (envelope and timing)
424 if (within >= WINDOW && within < symbol_blocks - 1) {
425 const float* power = ringPower(t);
426 for (int b = 0; b < data_bins; b++) c.acc[b] += power[b];
427 }
428 if (within == symbol_blocks - 1) endSymbol(c);
429 return;
430 }
431 // the end marker must follow the data
432 uint32_t end_start = (uint32_t)c.nsymbols * symbol_blocks;
433 uint32_t first = end_start + 2 * BLOCKS;
434 uint32_t last = end_start + (GGWaveConfig::MARKER_FRAMES - 2) * BLOCKS;
435 if (rel >= first && rel < last) c.end_score -= scores[t % SCORE_RING];
436 if (rel == last - 1) {
437 if (c.end_score >= 0.5f * detect_score * (last - first)) {
438 accept(c);
439 } else {
440 kill(c);
441 }
442 }
443 }
444
445 void endSymbol(Candidate& c) {
446 int bpt = c.protocol.bytes_per_tx;
447 for (int i = 0; i < 2 * bpt; i++) {
448 int best = 0;
449 float max_power = -1;
450 for (int k = 0; k < 16; k++) {
451 int b = 16 * i + k;
452 if (c.acc[b] > max_power) {
453 max_power = c.acc[b];
454 best = k;
455 }
456 }
457 uint8_t& value = c.bytes[c.symbol * bpt + i / 2];
458 if (i % 2 == 0) {
459 value = best;
460 } else {
461 value |= best << 4;
462 }
463 }
464 c.symbol++;
465 clearAccumulator(c);
466
467 if (c.nsymbols == 0 && c.symbol * bpt >= GGWaveConfig::LENGTH_BYTES) {
468 uint8_t tmp[GGWaveConfig::LENGTH_BYTES];
469 memcpy(tmp, c.bytes, sizeof(tmp));
470 if (!GGWaveRS::decode(tmp, 1, GGWaveConfig::LENGTH_BYTES - 1) ||
471 tmp[0] == 0 || tmp[0] > GGWaveConfig::MAX_PAYLOAD) {
472 kill(c);
473 return;
474 }
475 c.length = tmp[0];
476 int total = GGWaveConfig::encodedBytes(c.length);
477 c.nsymbols = (total + bpt - 1) / bpt;
478 }
479 if (c.nsymbols > 0 && c.symbol == c.nsymbols) {
480 uint8_t* data = c.bytes + GGWaveConfig::LENGTH_BYTES;
481 if (!GGWaveRS::decode(data, c.length, GGWaveConfig::eccBytes(c.length))) {
482 kill(c);
483 return;
484 }
485 c.data_done = true;
486 }
487 }

500 uint8_t* data = c.bytes + GGWaveConfig::LENGTH_BYTES;

508 if (cfg.rx_one) is_finished = true;
509 }
