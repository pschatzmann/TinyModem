60 bool begin() override {
61 is_active = false;
62 active_cfg = cfg;
63 ModemConfig& c = active_cfg;
64 if (!c.resolve()) {
65 LOGE("FSKDecoder: invalid configuration (max 64 bits per frame)");
66 return false;
67 }
68 if (!input.begin(c.bits_per_sample, c.channels, c.channel)) return false;
69 if (!plan.begin(c.sample_rate, c.mark_freq, c.space_freq, c.bandwidth)) {
70 LOGE("FSKDecoder: frequencies not supported with sample rate %d",
71 (int)c.sample_rate);
72 return false;
73 }
74 setupCodec();
75
76 nsamples_per_bit = c.sample_rate / c.baud_rate;
77
78 // For 8-bit frames with prev/start/stop we need 11 data-bits worth of
79 // samples, and we will scan through one bits worth at a time, hence we
80 // need a minimum total input buffer size of 12 data-bits.
81 unsigned nbits = 1 + c.start_bits + c.data_bits + 1;
82 size_t samplebuf_size = ceilf(nsamples_per_bit) * (nbits + 1);
83 samplebuf_size *= 2; // account for the half-buf filling method
84 if (c.buffer_divisor > 0 &&
85 samplebuf_size < c.sample_rate / (unsigned)c.buffer_divisor)
86 samplebuf_size = c.sample_rate / c.buffer_divisor;
87
88 // ensure that we overscan at least a single sample
89 nsamples_overscan = nsamples_per_bit * FRAME_OVERSCAN + 0.5f;
90 if (nsamples_overscan == 0) nsamples_overscan = 1;
91
92 frame_n_bits = c.frameBits();
93 frame_nsamples = nsamples_per_bit * frame_n_bits + 0.5f;
94
95 if (c.expect_bits != nullptr) {
96 strncpy(expect_data_string, c.expect_bits, MAX_EXPECT - 1);
97 expect_data_string[MAX_EXPECT - 1] = 0;
98 expect_n_bits = strlen(expect_data_string);
99 } else {
100 expect_n_bits = buildExpectBits(expect_data_string, false, 0);
101 }
102 if (c.rx_sync && c.sync_byte >= 0) {
103 buildExpectBits(expect_sync_string, true, c.sync_byte);
104 } else {
105 strcpy(expect_sync_string, expect_data_string);
106 }
107 expect_nsamples = nsamples_per_bit * expect_n_bits;
108
109 // we process the data when we have enough samples for the frame search
110 unsigned try_max_nsamples = nsamples_per_bit + nsamples_overscan;
111 size_t needed = try_max_nsamples + expect_nsamples +
112 ceilf(nsamples_per_bit) + 1;
113 process_threshold = samplebuf_size / 2;
114 if (process_threshold < needed) process_threshold = needed;
115 if (samplebuf_size < process_threshold) samplebuf_size = process_threshold;
116 samplebuf.resize(samplebuf_size);
117
118 resetState();
119 is_active = true;
120 return true;
121 }

263 int buildExpectBits(char* str, bool use_expect_bits, uint64_t expect_bits) {
264 ModemConfig& c = active_cfg;
265 char start_bit_value = c.invert_start_stop ? '1' : '0';
266 char stop_bit_value = c.invert_start_stop ? '0' : '1';
267 int j = 0;
268 if (c.stop_bits != 0.0f) str[j++] = stop_bit_value;
269 for (int i = 0; i < c.start_bits; i++) str[j++] = start_bit_value;
270 for (int i = 0; i < c.data_bits; i++, j++) {
271 str[j] = use_expect_bits ? ((expect_bits >> i) & 1) + '0' : 'd';
272 }
273 if (c.stop_bits != 0.0f) str[j++] = stop_bit_value;
274 str[j] = 0;
275 return j;
276 }

284 void addSample(float sample) {
285 if (is_finished) return;
286 if (skip > 0) {
287 skip--;
288 return;
289 }
290 if (active_cfg.rx_noise != 0.0f)
291 sample += noise() * active_cfg.rx_noise * 2.0f;
292 samplebuf[samples_nvalid++] = sample;
293 if (samples_nvalid >= process_threshold) {
294 while (step(false)) {
295 }
296 }
297 }
298
299 void applyAdvance() {
300 if (advance == 0) return;
301 if (advance >= samples_nvalid) {
302 skip = advance - samples_nvalid;
303 samples_nvalid = 0;
304 } else {
305 memmove(samplebuf.data(), samplebuf.data() + advance,
306 (samples_nvalid - advance) * sizeof(float));
307 samples_nvalid -= advance;
308 }
309 advance = 0;
310 }
311
314 bool step(bool eof) {
315 ModemConfig& c = active_cfg;
316 applyAdvance();
317 if (eof && skip > 0) return false;
318 if (!eof && samples_nvalid < process_threshold) return false;
319 if (samples_nvalid == 0) return false;
320 float* samples = samplebuf.data();
321 if (eof) {
322 // pad the incomplete data with silence
323 for (size_t j = samples_nvalid; j < samplebuf.size(); j++)
324 samples[j] = 0.0f;
325 }
326
327 // Auto-detect carrier frequency
328 if (c.auto_carrier && carrier_band < 0) {
329 unsigned i;
330 float nsamples_per_scan = nsamples_per_bit;
331 if (nsamples_per_scan > plan.fftSize()) nsamples_per_scan = plan.fftSize();
332 for (i = 0; i + nsamples_per_scan <= samples_nvalid;
333 i += nsamples_per_scan) {
334 carrier_band = plan.detectCarrier(samples + i, nsamples_per_scan,
335 c.auto_carrier_threshold);
336 if (carrier_band >= 0) break;
337 }
338 advance = i + nsamples_per_scan;
339 if (advance > samples_nvalid) advance = samples_nvalid;
340 if (carrier_band < 0) return true;
341
342 // default negative shift
343 float bw = plan.bandWidth();
344 int b_shift = -(float)(c.autodetect_shift + bw / 2.0f) / bw;
345 if (c.inverted) b_shift *= -1;
346 // only accept a carrier as b_mark if it will not result in a b_space
347 // band which is "too low".
348 int b_space = carrier_band + b_shift;
349 if (b_space < 1 || b_space >= (int)plan.bandCount()) {
350 carrier_band = -1;
351 return true;
352 }
353 plan.setTonesByBandShift(carrier_band, b_shift);
354 }
355
356 if (samples_nvalid < expect_nsamples) return false;
357
358 // try_max_nsamples
359 // 1. avoids finding a non-optimal first frame
360 // 2. allows us to track slightly slow signals
361 unsigned try_max_nsamples = carrier ? nsamples_per_bit * 0.75f + 0.5f
362 : nsamples_per_bit;
363 try_max_nsamples += nsamples_overscan;
364
365 unsigned try_step_nsamples = try_max_nsamples / FSK_ANALYZE_NSTEPS;
366 if (try_step_nsamples == 0) try_step_nsamples = 1;
367
368 float confidence, amplitude;
369 uint64_t bits = 0;
370 // frame_start_sample is the sample where the prev_stop bit begins
371 unsigned frame_start_sample = 0;
372 unsigned try_first_sample = carrier ? nsamples_overscan : 0;
373 float try_confidence_search_limit = c.search_limit;
374
375 confidence = plan.findFrame(
376 samples, expect_nsamples, try_first_sample, try_max_nsamples,
377 try_step_nsamples, try_confidence_search_limit,
378 carrier ? expect_data_string : expect_sync_string, &bits, &amplitude,
379 &frame_start_sample);
380
381 bool do_refine_frame = false;
382 if (confidence < peak_confidence * 0.75f) {
383 do_refine_frame = true;
384 peak_confidence = 0;
385 }
386
387 // no-confidence if amplitude drops abruptly to < 25% of the
388 // track_amplitude, which follows amplitude with hysteresis
389 if (amplitude < track_amplitude * 0.25f) confidence = 0;
390
391 if (confidence <= c.confidence) {
392 if (++noconfidence > FSK_MAX_NOCONFIDENCE_BITS) {
393 carrier_band = -1;
394 if (carrier) {
395 reportNoCarrier();
396 carrier = false;
397 carrier_nsamples = 0;
398 confidence_total = 0;
399 amplitude_total = 0;
400 nframes_decoded = 0;
401 track_amplitude = 0.0f;
402 if (c.rx_one) {
403 is_finished = true;
404 return false;
405 }
406 }
407 }
408 // continue searching from where we left off this time
409 advance = try_max_nsamples;
410 return true;
411 }
412
413 // Add a frame's worth of samples to the sample count
414 carrier_nsamples += frame_nsamples;
415
416 if (carrier) {
417 // If we already had carrier, adjust sample count +start -overscan
418 carrier_nsamples += frame_start_sample;
419 carrier_nsamples -= nsamples_overscan;
420 } else {
421 // We just acquired carrier.
422 reportCarrier();
423 carrier = true;
424 p_codec->reset();
425 do_refine_frame = true;
426 }
427
428 if (do_refine_frame && confidence < INFINITY && try_step_nsamples > 1) {
429 // Scan again, but try harder to find the best frame.
430 try_step_nsamples = try_max_nsamples / FSK_ANALYZE_NSTEPS_FINE;
431 if (try_step_nsamples == 0) try_step_nsamples = 1;
432 float confidence2, amplitude2;
433 uint64_t bits2;
434 unsigned frame_start_sample2;
435 confidence2 = plan.findFrame(
436 samples, expect_nsamples, try_first_sample, try_max_nsamples,
437 try_step_nsamples, INFINITY,
438 carrier ? expect_data_string : expect_sync_string, &bits2,
439 &amplitude2, &frame_start_sample2);
440 if (confidence2 > confidence) {
441 bits = bits2;
442 amplitude = amplitude2;
443 frame_start_sample = frame_start_sample2;
444 }
445 }
446
447 track_amplitude = (track_amplitude + amplitude) / 2;
448 if (peak_confidence < confidence) peak_confidence = confidence;
449 confidence_total += confidence;
450 amplitude_total += amplitude;
451 nframes_decoded++;
452 noconfidence = 0;
453
454 // Advance past the decoded frame, but a bit less than that to allow for
455 // tracking slightly fast signals, hence - nsamples_overscan.
456 advance = frame_start_sample + frame_nsamples - nsamples_overscan;
457
458 // chop off the prev_stop bit
459 if (c.stop_bits != 0.0f) bits = bits >> 1;
460 // chop off framing bits
461 bits = bitWindow(bits, c.start_bits, c.data_bits);
462 if (c.msb_first) bits = bitReverse(bits, c.data_bits);
463
464 // suppress printing of sync bytes
465 if (c.rx_sync && (long long)bits == c.sync_byte) return true;
466
467 if (p_print != nullptr) {
468 if (c.print_filter) {
469 print_filter.setOutput(p_print);
470 p_codec->decode(print_filter, bits, c.data_bits);
471 } else {
472 p_codec->decode(*p_print, bits, c.data_bits);
473 }
474 }
475 return true;
476 }

478 void reportCarrier() {
479 float rate = active_cfg.baud_rate;
480 if (rate >= 100)
481 status("### CARRIER %u @ %.1f Hz ###\n", (unsigned)(rate + 0.5f),
482 (double)markFrequency());
483 else
484 status("### CARRIER %.2f @ %.1f Hz ###\n", (double)rate,
485 (double)markFrequency());
486 }
487
488 void reportNoCarrier() {
489 float sample_rate = active_cfg.sample_rate;
490 float rate = active_cfg.baud_rate;
491 float nbits_decoded = nframes_decoded * frame_n_bits;
492 float throughput_rate =
493 nbits_decoded * sample_rate / (float)carrier_nsamples;
494 status("\n### NOCARRIER ndata=%u confidence=%.3f ampl=%.3f bps=%.2f",
495 nframes_decoded, (double)(confidence_total / nframes_decoded),
496 (double)(amplitude_total / nframes_decoded),
497 (double)throughput_rate);
498 if ((unsigned long long)(nbits_decoded * sample_rate + 0.5f) ==
499 (unsigned long long)(rate * carrier_nsamples)) {
500 status(" (rate perfect) ###\n");
501 } else {
502 float skew = (throughput_rate - rate) / rate;
503 status(" (%.1f%% %s) ###\n", (double)(fabsf(skew) * 100.0f),
504 signbit(skew) ? "slow" : "fast");
505 }
506 }
