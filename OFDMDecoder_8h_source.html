190 void addSample(float x, bool add_noise) {
191 if (is_finished) return;
192 if (add_noise && cfg.rx_noise != 0.0f)
193 x += noise() * cfg.rx_noise * 2.0f;
194 if (nvalid == buffer.size()) compact();
195 buffer[nvalid++] = x;
196 uint32_t t = now++;
197 if (filled <= (uint32_t)(2 * half)) filled++;
198
199 // sliding autocorrelation between the two halves of a window of
200 // fft_size samples which starts at d = t - fft_size + 1:
201 // corr = sum r(m - half) * r(m), power = sum r(m)^2 for the last half
202 power += x * x;
203 if (filled > (uint32_t)half) {
204 float a = sample(t - half);
205 corr += a * x;
206 power -= a * a;
207 if (filled > (uint32_t)(2 * half)) corr -= sample(t - 2 * half) * a;
208 }
209 if (filled >= (uint32_t)(2 * half)) {
210 if (t % 4096 == 0) recalculate(t);
211 uint32_t d = t - 2 * half + 1;
212 if (state == State::SEARCH) search(d);
213 }
214
215 while (state == State::FRAME && isReached(now, window + fft_size)) {
216 processSymbol();
217 }
218 }

223 void recalculate(uint32_t t) {
224 corr = power = 0;
225 for (int j = half - 1; j >= 0; j--) {
226 uint32_t m = t - j;
227 float b = sample(m);
228 corr += sample(m - half) * b;
229 power += b * b;
230 }
231 }

234 void compact() {
235 uint32_t keep = now - 2 * half - 1;
236 // the window of the sync symbol is in the plateau
237 if (in_plateau && !isReached(plateau_start, keep)) keep = plateau_start;
238 if (state == State::FRAME && !isReached(window, keep)) keep = window;
239 int32_t n = keep - base;
240 if (n <= 0) n = 1; // never overflow
241 memmove(buffer.data(), buffer.data() + n, (nvalid - n) * sizeof(float));
242 nvalid -= n;
243 base += n;
244 }

248 void search(uint32_t d) {
249 float metric = power > MIN_POWER * half ? corr / power : 0;
250 if (metric > cfg.sync_threshold) {
251 if (!in_plateau) {
252 in_plateau = true;
253 plateau_start = d;
254 plateau_len = 0;
255 }
256 // the plateau of a sync symbol is about as long as the cyclic prefix:
257 // a much longer one is a periodic signal (e.g. a steady tone), which
258 // would prevent that the samples of the plateau are dropped
259 if (plateau_len == (int)plateau.size()) {
260 in_plateau = false;
261 return;
262 }
263 plateau[plateau_len++] = metric;
264 return;
265 }
266 if (!in_plateau) return;
267 in_plateau = false;
268 // smooth the metric against noise and use the middle of the range with
269 // at least 80% of the peak
270 int w = prefix_size / 4;
271 if (w < 1) w = 1;
272 float peak = 0;
273 for (int j = 0; j < plateau_len; j++) {
274 float sum = 0;
275 int n = 0;
276 for (int k = j - w / 2; k <= j + w / 2; k++) {
277 if (k < 0 || k >= plateau_len) continue;
278 sum += plateau[k];
279 n++;
280 }
281 smoothed[j] = sum / n;
282 if (smoothed[j] > peak) peak = smoothed[j];
283 }
284 int first = -1, last = -1;
285 for (int j = 0; j < plateau_len; j++) {
286 if (smoothed[j] >= 0.8f * peak) {
287 if (first < 0) first = j;
288 last = j;
289 }
290 }
291 if (first < 0 || last - first < prefix_size / 4) return;
292 window = plateau_start + (first + last) / 2;
293 state = State::FRAME;
294 symbol = 0;
295 drift = 0;
296 slope_adjust = 0;
297 }

300 void dft() {
301 for (int i = 0; i < ncarriers; i++) {
302 int bin = first_bin + i;
303 float re = 0, im = 0;
304 int idx = 0;
305 for (int n = 0; n < fft_size; n++) {
306 float x = sample(window + n);
307 re += x * table.cosine(idx);
308 im -= x * table.sine(idx);
309 idx += bin;
310 if (idx >= fft_size) idx -= fft_size;
311 }
312 z_re[i] = re;
313 z_im[i] = im;
314 }
315 }

317 static float wrap(float angle) {
318 while (angle > (float)M_PI) angle -= 2.0f * (float)M_PI;
319 while (angle <= -(float)M_PI) angle += 2.0f * (float)M_PI;
320 return angle;
321 }

324 static void fit(const float* angle, const float* weight, int n,
325 float& common, float& slope) {
326 float sw = 0, sx = 0, sy = 0, sxx = 0, sxy = 0;
327 for (int i = 0; i < n; i++) {
328 float w = weight[i];
329 if (w <= 0) continue;
330 sw += w;
331 sx += w * i;
332 sy += w * angle[i];
333 sxx += w * i * i;
334 sxy += w * i * angle[i];
335 }
336 float det = sw * sxx - sx * sx;
337 if (sw <= 0 || fabsf(det) < 1e-12f) {
338 common = sw > 0 ? sy / sw : 0;
339 slope = 0;
340 return;
341 }
342 slope = (sw * sxy - sx * sy) / det;
343 common = (sy - slope * sx) / sw;
344 }

346 void processSymbol() {
347 dft();
348 if (symbol == 0) {
349 // sync symbol: keep it for the reference symbol
350 for (int i = 0; i < ncarriers; i++) {
351 prev_re[i] = z_re[i];
352 prev_im[i] = z_im[i];
353 }
354 } else if (symbol == 1) {
355 referenceSymbol();
356 } else {
357 dataSymbol();
358 }
359 if (state != State::FRAME) return;
360 symbol++;
361 // timing: keep the window in the middle of the cyclic prefix
362 window += symbol_size;
363 slope_adjust = 0;
364 if (drift >= 1.0f) {
365 window += 1;
366 drift -= 1.0f;
367 slope_adjust = 2.0f * (float)M_PI / fft_size;
368 } else if (drift <= -1.0f) {
369 window -= 1;
370 drift += 1.0f;
371 slope_adjust = -2.0f * (float)M_PI / fft_size;
372 }
373 }

377 void referenceSymbol() {
378 float angle[128], weight[128];
379 reference_power = 0;
380 for (int i = 0; i < ncarriers; i++) {
381 float power_i = z_re[i] * z_re[i] + z_im[i] * z_im[i];
382 reference_power += power_i;
383 weight[i] = 0;
384 angle[i] = 0;
385 if ((first_bin + i) & 1) continue;
386 // z * conj(prev) rotated by the known phase difference
387 float dr = z_re[i] * prev_re[i] + z_im[i] * prev_im[i];
388 float di = z_im[i] * prev_re[i] - z_re[i] * prev_im[i];
389 int known = OFDMConfig::knownPhase(1, i) - OFDMConfig::knownPhase(0, i);
390 angle[i] = wrap(atan2f(di, dr) - known * (float)M_PI / 2);
391 weight[i] = sqrtf(dr * dr + di * di);
392 }
393 reference_power /= ncarriers;
394 fit(angle, weight, ncarriers, pred_common, pred_slope);
395 drift = 0;
396 for (int b = 0; b < 32; b++) header_soft[b] = 0;
397 header_symbol = 0;
398 frame_len = -1;
399 ndata = 0;
400 current = 0;
401 current_dibits = 0;
402 error_total = 0;
403 error_count = 0;
404 for (int i = 0; i < ncarriers; i++) {
405 prev_re[i] = z_re[i];
406 prev_im[i] = z_im[i];
407 }
408 }

410 void dataSymbol() {
411 float angle[128], weight[128];
412 float symbol_power = 0;
413 int dibits[128];
414 bool is_header = frame_len < 0;
415 // header: 0 or 180 degrees, data: 0, 90, 180 or 270 degrees
416 float step = is_header ? (float)M_PI : (float)M_PI / 2;
417 for (int i = 0; i < ncarriers; i++) {
418 symbol_power += z_re[i] * z_re[i] + z_im[i] * z_im[i];
419 float dr = z_re[i] * prev_re[i] + z_im[i] * prev_im[i];
420 float di = z_im[i] * prev_re[i] - z_re[i] * prev_im[i];
421 float a = wrap(atan2f(di, dr) - pred_common -
422 (pred_slope + slope_adjust) * i);
423 int q = lroundf(a / step);
424 angle[i] = a - q * step; // residual error
425 weight[i] = sqrtf(dr * dr + di * di);
426 if (is_header) {
427 // soft decision: sum of all copies of the header bit
428 header_soft[(header_symbol * ncarriers + i) % 32] += cosf(a);
429 } else {
430 dibits[i] = phaseToGray(q & 3);
431 }
432 prev_re[i] = z_re[i];
433 prev_im[i] = z_im[i];
434 error_total += fabsf(angle[i]);
435 error_count++;
436 }
437 // the signal is lost
438 if (symbol_power / ncarriers < 0.1f * reference_power) {
439 endFrame(false);
440 return;
441 }
442 // track the frequency offset and the timing drift
443 float common, slope;
444 fit(angle, weight, ncarriers, common, slope);
445 pred_common += TRACKING_GAIN * common;
446 pred_slope += TRACKING_GAIN * slope;
447 // timing drift in samples per symbol
448 drift -= pred_slope * fft_size / (2.0f * (float)M_PI);
449
450 if (is_header) {
451 if (++header_symbol == OFDMConfig::headerSymbols(ncarriers))
452 checkHeader();
453 return;
454 }
455 for (int i = 0; i < ncarriers && state == State::FRAME; i++)
456 dataBits(dibits[i]);
457 }

461 static const int dibit[4] = {0, 1, 3, 2};

465 void checkHeader() {
466 uint32_t header = 0;
467 for (int b = 0; b < 32; b++) header = (header << 1) | (header_soft[b] < 0);
468 uint16_t len = header >> 16;
469 uint16_t check = header & 0xFFFF;
470 if ((uint16_t)~len != check || len == 0) {
471 // not a valid frame: search again
472 state = State::SEARCH;
473 in_plateau = false;
474 return;
475 }
476 frame_len = len;
477 status("### CARRIER ofdm%d %.1f-%.1f Hz len=%d ###\n", ncarriers,
478 (double)cfg.carrierFrequency(0),
479 (double)cfg.carrierFrequency(ncarriers - 1), frame_len);
480 }
481
482 void dataBits(int dibit) {
483 current = (current << 2) | dibit;
484 if (++current_dibits == 4) {
485 if (p_print != nullptr) p_print->write(current);
486 ndata++;
487 current = 0;
488 current_dibits = 0;
489 if (ndata == frame_len) endFrame(true);
490 }
491 }
492
493 void endFrame(bool complete) {
494 state = State::SEARCH;
495 in_plateau = false;
496 if (frame_len < 0) return; // no valid header
497 if (complete) frames++;
498 float error = error_count > 0 ? error_total / error_count : 0;
499 status("\n### NOCARRIER ndata=%d phase_error=%.1f deg%s ###\n", ndata,
500 (double)(error * 180.0f / (float)M_PI),
501 complete ? "" : " (incomplete)");
502 frame_len = -1;
503 if (cfg.rx_one) is_finished = true;
504 }
