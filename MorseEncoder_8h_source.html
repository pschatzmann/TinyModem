137 void sendCode(const char* code) {
138 const MorseCode::Timing& t = MorseCode::timing(cfg.alphabet);
139 if (!transmitting) {
140 transmitting = true;
141 position = written;
142 } else {
143 // the letter space was sent after the previous character
144 if (word_pending) silence(t.word_gap - t.letter_gap);
145 }
146 word_pending = false;
147 for (const char* c = code; *c; c++) {
148 if (*c == ' ') {
149 // space within an American character instead of the element space
150 silence(t.inner_gap - t.element_gap);
151 continue;
152 }
153 float len = *c == '.' ? t.dot
154 : *c == '-' ? t.dash
155 : *c == '_' ? t.long_dash
156 : t.extra_long_dash;
157 tone(len);
158 silence(c[1] ? t.element_gap : t.letter_gap);
159 }
160 }

162 void silence(float units) {

165 }

168 void tone(float units) {
169 position += units * unit_samples;
170 uint64_t start = written, stop = llround(position);
171 size_t n = stop - start;
172 size_t ramp = cfg.ramp_ms * cfg.sample_rate / 1000;
173 if (2 * ramp > n) ramp = n / 2;
174 re = 1;
175 im = 0;
176 for (size_t j = 0; j < n; j++) {
177 float gain = 1.0f;
178 if (j < ramp) gain = 0.5f - 0.5f * cosf((float)M_PI * j / ramp);
179 if (n - j <= ramp) gain = 0.5f - 0.5f * cosf((float)M_PI * (n - j) / ramp);
180 samples.write(gain * im);
181 float r = re * step_re - im * step_im;
182 im = re * step_im + im * step_re;
183 re = r;
184 if (++sample_count % 1024 == 0) {
185 float mag = sqrtf(re * re + im * im);
186 re /= mag;
187 im /= mag;
188 }
189 }
190 written = stop;
191 }
